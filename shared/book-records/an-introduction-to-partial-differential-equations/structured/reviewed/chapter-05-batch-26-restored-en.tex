\paragraph*{Problems}

\begin{enumerate}[label=\thesection.\arabic*.,ref=\thesection.\arabic*,start=45,series=book-problems]

\item\label{ex:compact-fourier-compact-support} Let $f\in\mathcal{D}(\mathbb{R}^m)$. Under what conditions is $\mathcal{F}[f]$ also in $\mathcal{D}(\mathbb{R}^m)$? Hint: Consider $\mathcal{F}[f]$ as a function of a complex argument.

\item\label{ex:fourier-gaussian-rational-transforms} Find the Fourier transforms of the following functions: $\exp(-x^2)$, $1/(1+x^2)$, $\sin x/(1+x^2)$.

\item\label{ex:tempered-fourier-differentiation-reflection} Check that \eqref{eq:fourier-frequency-derivatives}, \eqref{eq:fourier-spatial-derivatives} and \eqref{eq:fourier-transform-twice-reflection} hold for tempered distributions.

\item\label{ex:fourier-polynomial-growth-regularizations} Find the Fourier transforms of $|x|$, $\sin(x^2)$, $x_+^{1/2}$. Hint: Try modifying the functions using factors like $e^{-\varepsilon x^2}$ and passing to the limit.

\item\label{ex:fourier-linear-change-radial-symmetry} Let $\bm{A}$ be a nonsingular matrix. How is the Fourier transform of $f(\bm{A}\bm{x})$ related to that of $f(\bm{x})$? Use the result to show that the Fourier transform of a radially symmetric tempered distribution is radially symmetric.

\item\label{ex:fourier-heat-fundamental-solution} Use the Fourier transform to find the fundamental solution for the heat equation.

\item\label{ex:fourier-laplace-fundamental-three-dimensions} Use the Fourier transform to find the fundamental solution for Laplace\textquotesingle s equation in $\mathbb{R}^3$.

\item\label{ex:one-sided-tempered-fourier-product} (a) Let $f\in\mathcal{S}^{\prime}(\mathbb{R})$ and assume that the support of $f$ is contained in $\{x\geq0\}$. Show that the Fourier transform of $f$ is an analytic function in the half-plane $\operatorname{Im}\xi<0$.

(b) Let $f,g\in\mathcal{S}^{\prime}(\mathbb{R})$ have support in $\{x\geq0\}$. Show that $f*g$ also has support in $\{x\geq0\}$ and that \eqref{eq:fourier-convolution-product-test-functions} holds (in the pointwise sense) in the lower half-plane.

\item\label{ex:laplace-one-dimensional-heat-fundamental} Use the Laplace transform to find the fundamental solution of the heat equation in one space dimension.

\item\label{ex:laplace-heat-boundary-smoothness} Show that, for any $t>0$, \eqref{eq:heat-unit-boundary-inverse-laplace-formula} represents a $C^\infty$ function of $x$ for $x\in[0,1]$. Hint: First deform the contour into the left half-plane. Then differentiate under the integral.

\item\label{ex:backwards-heat-laplace-failure} In \eqref{eq:heat-unit-boundary-zero-initial-problem}, replace the heat equation by the backwards heat equation $u_t=-u_{xx}$. Explain what goes wrong when you try to solve the problem by Laplace transform.

\end{enumerate}
