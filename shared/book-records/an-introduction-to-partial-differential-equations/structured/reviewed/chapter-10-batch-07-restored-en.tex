\Needspace{5\baselineskip}\textbf{Problems}

\begin{enumerate}[label=\thesection.\arabic*.,ref=\thesection.\arabic*,start=1,series=book-problems]

\item\label{ex:banach-contraction-picard-lindelof-existence} Apply the Banach contraction principle to prove the Picard-Lindelöf existence theorem for ODEs (Theorem~\ref{thm:ode-existence}).

\item\label{ex:continuous-bilinear-linear-operator-correspondence} Show that there is a natural correspondence between continuous bilinear mappings from $X$ to $Y$ and linear mappings from $X$ to $\mathcal{L}(X,Y)$.

\item\label{ex:frechet-derivative-chain-rule} Establish the chain rule for Fréchet derivatives.

\item\label{ex:pointwise-composition-frechet-derivative} Verify the claim in Example~\ref{ex:pointwise-composition-frechet-derivative-example}.

\item\label{ex:inverse-function-higher-differentiability} Verify Remark~\ref{rem:inverse-function-higher-differentiability}.

\item\label{ex:sobolev-composition-lemma-details} Complete the details in the proof of Lemma~\ref{lem:sobolev-pointwise-composition-differentiability}.

\item\label{ex:implicit-function-fully-nonlinear-elliptic-boundary-problems} Use the implicit function theorem to obtain existence results for fully nonlinear second-order elliptic PDEs with nonlinear boundary conditions.

\item\label{ex:sine-composition-ltwo-nowhere-differentiability} Consider the composition mapping\[L^2(0,1)\ni u(\cdot)\mapsto\sin(u(\cdot))\in L^2(0,1).\]Show that this mapping is nowhere differentiable.

\end{enumerate}
