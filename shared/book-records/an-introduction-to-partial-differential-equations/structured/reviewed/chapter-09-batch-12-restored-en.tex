\Needspace{5\baselineskip}\textbf{Problems}

\begin{enumerate}[label=\thesection.\arabic*.,ref=\thesection.\arabic*,start=1,series=book-problems]

\item\label{ex:punctured-disk-dirichlet-weak-solution} Let $D$ be the unit disk in the plane and let $\Omega=D\backslash\{0\}$. It is well-known that the Dirichlet problem $\Delta u=1$ with $u=0$ on $\partial\Omega$ has no classical solution. What is the weak ``solution\textquotesingle\textquotesingle\ given by Theorem~\ref{thm:weak-dirichlet-positive-shift-existence}? Hint: First characterize $H_0^1(\Omega)$.

\item\label{ex:ode-coercive-dirichlet-uniqueness} Consider the ODE boundary-value problem $y''+p(x)y'+q(x)y=f(x)$, $y(0)=y(1)=0$. Here $p\in C^1[0,1]$, $q\in C[0,1]$. Prove that a unique solution exists if $p'-2q\geq0$.

\item\label{ex:positive-double-sequence-l2-solvability} Let the double sequence $a_{ij}$ be such that $\sum_{i,j=1}^\infty|a_{ij}|^2<\infty$. Assume, moreover, that the matrix $a_{ij}$, $i,j=1,\ldots,N$, is positive definite for every $N$. Prove that the equation
\begin{equation}\label{eq:positive-double-sequence-l2-system-problem}
u_n+\sum_{j=1}^\infty a_{nj}u_j=f_n
\end{equation}
has a unique solution $u\in\ell^2$ for every $f\in\ell^2$.

\item\label{ex:discontinuous-coefficient-interface-matching} Consider a ``weak\textquotesingle\textquotesingle\ solution of the Dirichlet problem for the differential operator defined in \eqref{eq:second-order-garding-divergence-operator} in a situation where the coefficients $a_{ij}$, $b_i$ and $c$ have discontinuities across a smooth surface. Assume you know that the solution is smooth on both sides of this interface. Determine the ``matching conditions\textquotesingle\textquotesingle\ which are satisfied across the interface.

\end{enumerate}
