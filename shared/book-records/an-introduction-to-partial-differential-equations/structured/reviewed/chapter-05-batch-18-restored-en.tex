\begin{Example}{The wave equation}
\label{ex:wave-fundamental-solution}
For second-order equations
\begin{equation}\label{eq:second-order-time-evolution-fundamental-equation}
u_{tt}=L(D)u
\end{equation}
we define the fundamental solution $G$ as a twice continuously differentiable function $[0,\infty)\to\mathcal{D}^{\prime}(\mathbb{R}^m)$ such that $G$ satisfies \eqref{eq:second-order-time-evolution-fundamental-equation} with the initial conditions $G(\bm{x},0)=0$, $G_t(\bm{x},0)=\delta(\bm{x})$. The solution of the inhomogeneous problem
\begin{equation}\label{eq:second-order-evolution-inhomogeneous-initial-problem}
u_{tt}=L(D)u+f(\bm{x},t),\quad u(\bm{x},0)=u_0(\bm{x}),\quad u_t(\bm{x},0)=u_1(\bm{x})
\end{equation}
is then represented by
\begin{equation}\label{eq:second-order-evolution-fundamental-solution-formula}
\begin{aligned}
u(\bm{x},t)&=\int_{\mathbb{R}^m}G_t(\bm{x}-\bm{y},t)u_0(\bm{y})\,\mathrm{d}\bm{y}
+\int_{\mathbb{R}^m}G(\bm{x}-\bm{y},t)u_1(\bm{y})\,\mathrm{d}\bm{y}\\
&\quad+\int_0^t\int_{\mathbb{R}^m}G(\bm{x}-\bm{y},t-s)f(\bm{y},s)\,\mathrm{d}\bm{y}\,\mathrm{d}s.
\end{aligned}
\end{equation}
For the wave equation in one space dimension,
\begin{equation}\label{eq:wave-one-dimensional-fundamental-solution}
G(x,t)=\begin{cases}1/2&|x|<t\\0&|x|\geq t\end{cases}
\end{equation}
is a fundamental solution. Indeed, it is obvious that $G(x,0)=0$ and from the representation $G(x,t)=(H(x+t)-H(x-t))/2$ it follows that $G$ satisfies the wave equation and that $G_t(x,t)=(\delta(x+t)+\delta(x-t))/2$, which equals $\delta(x)$ for $t=0$. The fundamental solution for the wave equation in several dimensions will be discussed in the next section. We draw attention to the fact that the fundamental solutions for the Laplace and heat equations are (apart from the singularity at the origin) $C^\infty$-functions, but that of the wave equation is not. This has important implications for the regularity of solutions.
\end{Example}

\paragraph*{Problems}

\begin{enumerate}[label=\thesection.\arabic*.,ref=\thesection.\arabic*,start=33,series=book-problems]

\item\label{ex:distribution-smooth-test-parameter-map} Let $\phi(\bm{x},\bm{y})\in\mathcal{D}(\mathbb{R}^{p+q})$, $g\in\mathcal{D}^{\prime}(\mathbb{R}^q)$. Show that \[\psi(\bm{x}):=(g(\bm{y}),\phi(\bm{x},\bm{y}))\] is in $\mathcal{D}(\mathbb{R}^p)$.

\item\label{ex:tempered-direct-product} Show that the direct product can also be defined in $\mathcal{S}^{\prime}$.

\item\label{ex:direct-product-support} Let $F$ and $G$ be the supports of $f$ and $g$. Show that the support of the direct product is $F\times G$.

\item\label{ex:test-function-integral-riemann-sums} Let $\phi,\psi\in\mathcal{D}(\mathbb{R}^m)$. Show that $\int_{\mathbb{R}^m}\psi(\bm{y}-\bm{x})\phi(\bm{y})\,\mathrm{d}\bm{y}$ defines an element of $\mathcal{D}(\mathbb{R}^m)$. Moreover, show that, in the sense of convergence in $\mathcal{D}(\mathbb{R}^m)$, the integral is a limit of Riemann sums.

\item\label{ex:distribution-convolution-continuity} Prove Lemma~\ref{lem:distribution-convolution-continuity}.

\item\label{ex:test-functions-dense-tempered-distributions} Discuss how the proof of Theorem~\ref{thm:test-functions-dense-distributions} needs to be modified to show that $\mathcal{D}(\Omega)$ is dense in $\mathcal{D}^{\prime}(\Omega)$. Also show that $\mathcal{D}(\mathbb{R}^m)$ is dense in $\mathcal{S}^{\prime}(\mathbb{R}^m)$.

\item\label{ex:direct-product-joint-continuity} Show that the direct product is jointly continuous in its factors.

\item\label{ex:biharmonic-fundamental-solution} Find a fundamental solution for the biharmonic equation $\Delta\Delta G=\delta(\bm{x})$ on $\mathbb{R}^m$.

\item\label{ex:inhomogeneous-fundamental-solution-formula} Prove that \eqref{eq:first-order-evolution-fundamental-solution-formula} is indeed a solution of \eqref{eq:first-order-evolution-inhomogeneous-initial-problem}.

\item\label{ex:evolution-fundamental-spacetime-delta} Let $G$ be the fundamental solution corresponding to the initial-value problem of $u_t=L(D)u$. Show that the functional
\begin{equation}\label{eq:evolution-fundamental-solution-spacetime-functional}
F:\phi\mapsto\int_0^\infty\int_{\mathbb{R}^m}G(\bm{x},t)\phi(\bm{x},t)\,\mathrm{d}\bm{x}\,\mathrm{d}t
\end{equation}
defines a distribution on $\mathbb{R}^{m+1}$ and that this distribution satisfies the equation $F_t-L(D)F=\delta(\bm{x},t)$.

\item\label{ex:wave-one-dimensional-duhamel-formula} Specialize \eqref{eq:second-order-evolution-fundamental-solution-formula} to the one-dimensional wave equation.

\item\label{ex:polynomial-compact-distribution-convolution} Let $f$ be a distribution with compact support and let $P$ be a polynomial. Show that $P*f$ is a polynomial.

\end{enumerate}
