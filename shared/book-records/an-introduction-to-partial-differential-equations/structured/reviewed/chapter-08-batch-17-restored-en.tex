\Needspace{5\baselineskip}\textbf{Problems}

\begin{enumerate}[label=\thesection.\arabic*.,ref=\thesection.\arabic*,start=17,series=book-problems]

\item\label{ex:orthogonal-projection-spectrum} Describe the spectrum $\sigma(P_M)$ of the projection operator described in Example~\ref{ex:orthogonal-projection-operator}.

\item\label{ex:continuous-multiplication-operator-spectrum} \begin{enumerate}[label=(\alph*)]\item Define a multiplication operator $M:C_b([0,1])\to C_b([0,1])$ by\[M(u)(x):=xu(x),\]for every $u\in C_b([0,1])$. Describe $\sigma(M)$.\item Let $v\in C_b([0,1])$ be given. Define an operator $M_v:C_b([0,1])\to C_b([0,1])$ by\[M_v(u)(x):=v(x)u(x),\]for every $u\in C_b([0,1])$. Describe $\sigma(M_v)$.\end{enumerate}

\item\label{ex:bounded-operator-extension-spectral-inclusions} Suppose that $(\mathcal{D}(\widetilde A),\widetilde A)$ is an extension of a bounded operator $(\mathcal{D}(A),A)$. Show the following:\begin{enumerate}[label=(\alph*)]
\item $\sigma_p(\widetilde A)\supset\sigma_p(A)$.
\item $\sigma_r(\widetilde A)\subset\sigma_r(A)$.
\item $\sigma_c(A)\subset\sigma_c(\widetilde A)\cup\sigma_p(\widetilde A)$.
\item $\rho(\widetilde A)\subset\rho(A)\cup\sigma_r(A)$.
\end{enumerate}

\item\label{ex:bounded-operator-resolvent-decay-at-infinity} Let $A\in\mathcal{L}(X)$. Show that $\|R_\lambda(A)\|\to0$ as $|\lambda|\to\infty$.

\item\label{ex:unbounded-second-derivative-noncompact-spectrum} Let\[\mathcal{D}(A)=\{u\in H^2(0,1)\mid u(0)=u(1)=0\}.\]Define the operator $(\mathcal{D}(A),A)$ from $L^2(0,1)$ to $L^2(0,1)$ by\[Au=u''\]for $u\in\mathcal{D}(A)$. Show that $\sigma(A)$ is not compact. Does your answer contradict Corollary~\ref{cor:bounded-operator-spectrum-compact}?

\item\label{ex:weak-and-strong-banach-valued-analyticity} Let $G\subset\mathbb{C}$ be a domain and let $X$ be a Banach space. Then a mapping
\begin{equation}\label{eq:weakly-analytic-banach-valued-map}
\mathbb{C}\supset G\ni\lambda\mapsto B(\lambda)\in X
\end{equation}
is said to be \textbf{weakly analytic} at $\lambda_0\in\mathbb{C}$ if, for every $g\in X^*$, the complex-valued function defined by
\begin{equation}\label{eq:weakly-analytic-scalar-functional-composition}
f(\lambda):=g(B(\lambda))
\end{equation}
is analytic (in the usual sense) in a neighborhood of $\lambda_0$. The function $B(\lambda)$ is analytic on $G$ if it is analytic at each point in $G$.\begin{enumerate}[label=(\alph*)]\item Show that (strong) analyticity implies weak analyticity.\item Show that weak analyticity implies (strong) analyticity.\end{enumerate}

\item\label{ex:closed-unbounded-resolvent-analyticity} Extend Theorems~\ref{thm:local-resolvent-power-series} and \ref{thm:bounded-operator-analytic-resolvent} to unbounded closed operators.

\end{enumerate}
