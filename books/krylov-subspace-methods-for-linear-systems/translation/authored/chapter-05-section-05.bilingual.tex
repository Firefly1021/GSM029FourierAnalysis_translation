\subsection{\BT{Matrix Trigonometric Functions}{矩阵三角函数}}\label{sec:matrix-trigonometric-functions}

% block: scalar-trigonometric-expansions page:214 type:ordinary-paragraph
\BT{Here we consider matrix trigonometric functions: matrix sine function $\sin(A)$ and matrix cosine function $\cos(A)$. $\sin(x)$ and $\cos(x)$ can be expanded as}{这里考虑矩阵三角函数: 矩阵正弦函数 $\sin(A)$ 和矩阵余弦函数 $\cos(A)$. $\sin(x)$ 与 $\cos(x)$ 可展开为}
\[\begin{aligned}\sin(x)&=x-\frac{1}{3!}x^3+\frac{1}{5!}x^5-\frac{1}{7!}x^7+\cdots,\\\cos(x)&=1-\frac{1}{2!}x^2+\frac{1}{4!}x^4-\frac{1}{6!}x^6+\cdots.\end{aligned}\]

% block: matrix-trigonometric-expansions pages:214,215 type:ordinary-paragraph
\BT{$\sin(A)$ and $\cos(A)$ are defined for any $A\in\mathbb{C}^{n\times n}$ by using the above expansions as described below:}{利用上述展开, 对任意 $A\in\mathbb{C}^{n\times n}$, 如下定义 $\sin(A)$ 与 $\cos(A)$:}
\[\begin{aligned}\sin(A)&=A-\frac{1}{3!}A^3+\frac{1}{5!}A^5-\frac{1}{7!}A^7+\cdots,\\\cos(A)&=I-\frac{1}{2!}A^2+\frac{1}{4!}A^4-\frac{1}{6!}A^6+\cdots.\end{aligned}\]

% block: matrix-trigonometric-properties-transition page:215 type:ordinary-paragraph
\BT{Below are the fundamental properties of $\sin(A)$ and $\cos(A)$.}{下面给出 $\sin(A)$ 与 $\cos(A)$ 的基本性质.}

% block: matrix-trigonometric-properties-theorem page:215 type:theorem
\Needspace{14\baselineskip}
\begin{Theorem}{\BT{Some properties of $\sin(A)$ and $\cos(A)$}{$\sin(A)$ 与 $\cos(A)$ 的一些性质}}\label{thm:matrix-trigonometric-properties}
\BT{The following facts hold true for $A,B\in\mathbb{C}^{n\times n}$:}{对 $A,B\in\mathbb{C}^{n\times n}$, 下列事实成立:}
\begin{enumerate}[label=(\arabic*)]
\item\label{part:matrix-trig-euler} $e^{iA}=\cos(A)+i\sin(A)$;
\item\label{part:matrix-sine-exponentials} $\sin(A)=(e^{iA}-e^{-iA})/2i$;
\item\label{part:matrix-cosine-exponentials} $\cos(A)=(e^{iA}+e^{-iA})/2$;
\item $\sin(-A)=-\sin(A)$, $\cos(-A)=\cos(A)$;
\item $\sin(A)^2+\cos(A)^2=I$;
\item \BT{$\sin(A\pm B)=\sin(A)\cos(B)\pm\cos(A)\sin(B)$ if $AB=BA$;}{如果 $AB=BA$, 则\newline{}$\sin(A\pm B)=\sin(A)\cos(B)\pm\cos(A)\sin(B)$;}
\item \BT{$\cos(A\pm B)=\cos(A)\cos(B)\mp\sin(A)\sin(B)$ if $AB=BA$.}{如果 $AB=BA$, 则\newline{}$\cos(A\pm B)=\cos(A)\cos(B)\mp\sin(A)\sin(B)$.}
\end{enumerate}
\end{Theorem}

% block: matrix-trigonometric-computational-derivation page:215 type:ordinary-paragraph
\BT{\ref{part:matrix-sine-exponentials} and \ref{part:matrix-cosine-exponentials} of Theorem~\ref{thm:matrix-trigonometric-properties} can be derived from \ref{part:matrix-trig-euler} of Theorem~\ref{thm:matrix-trigonometric-properties}. From \ref{part:matrix-sine-exponentials} and \ref{part:matrix-cosine-exponentials}, $\sin(A)$ and $\cos(A)$ can be obtained by matrix exponential functions. If $A$ is a real matrix, it follows from \ref{part:matrix-trig-euler} of Theorem~\ref{thm:matrix-trigonometric-properties} that $\cos(A)$ corresponds to the real part of $e^{iA}$, and $\sin(A)$ corresponds to the imaginary part of $e^{iA}$. From this, Algorithm~\ref{alg:matrix-sine-cosine} is obtained for computing $\sin(A)$ and $\cos(A)$ as described in Algorithm~\ref{alg:matrix-sine-cosine}.}{Theorem~\ref{thm:matrix-trigonometric-properties} 的第 \ref{part:matrix-sine-exponentials} 与 \ref{part:matrix-cosine-exponentials} 项可由其第 \ref{part:matrix-trig-euler} 项导出. 由第 \ref{part:matrix-sine-exponentials} 和 \ref{part:matrix-cosine-exponentials} 项, 可通过矩阵指数函数得到 $\sin(A)$ 与 $\cos(A)$. 如果 $A$ 是实矩阵, 则由 Theorem~\ref{thm:matrix-trigonometric-properties} 的第 \ref{part:matrix-trig-euler} 项可知, $\cos(A)$ 对应于 $e^{iA}$ 的实部, $\sin(A)$ 对应于 $e^{iA}$ 的虚部. 由此得到计算 $\sin(A)$ 与 $\cos(A)$ 的 Algorithm~\ref{alg:matrix-sine-cosine}, 如 Algorithm~\ref{alg:matrix-sine-cosine} 所示.}

% block: matrix-sine-cosine-algorithm page:215 type:ordinary-paragraph
\begin{algorithm}[H]\caption{\BT{Computation of $S=\sin(A)$ and $C=\cos(A)$}{计算 $S=\sin(A)$ 和 $C=\cos(A)$}}\label{alg:matrix-sine-cosine}\begin{algorithmic}[1]
\State \BT{Compute $X=e^{iA}$.}{计算 $X=e^{iA}$.}
\If{\BT{$A$ is a real matrix}{$A$ 是实矩阵}}
\State $S=\operatorname{Im}(X)$ (\BT{imaginary part of $A$}{$A$ 的虚部}), $C=\operatorname{Re}(X)$ (\BT{real part of $A$}{$A$ 的实部})
\EndIf
\If{\BT{$A$ is a complex matrix}{$A$ 是复矩阵}}
\State $S=\frac{1}{2i}(X-X^{-1})$, $C=\frac{1}{2}(X+X^{-1})$
\EndIf
\end{algorithmic}\end{algorithm}
