\Needspace{5\baselineskip}\textbf{Problems}

\begin{enumerate}[label=\thesection.\arabic*.,ref=\thesection.\arabic*,start=48,series=book-problems]

\item\label{ex:sturm-liouville-neumann-robin-spectrum} 求下列边值问题的特征值和特征函数:\[\begin{aligned}
-u''-\lambda u&=0,\\
u'(0)&=0,\\
\cos\beta\,u(1)+\sin\beta\,u'(1)&=0.
\end{aligned}\]

\item\label{ex:sturm-liouville-weight-x-dirichlet-spectrum} 求下列边值问题的特征值和特征函数:\[\begin{aligned}
-(xu')'-\lambda xu&=0,\quad0<a<x<b;\\
u(a)&=0,\\
u(b)&=0.
\end{aligned}\]

\item\label{ex:sturm-liouville-general-energy-bound} 在 $\tan\alpha\in(-\infty,0)$ 的情形下证明 \eqref{eq:sturm-liouville-lower-energy-bound}.

\item\label{ex:sturm-liouville-spectral-sobolev-characterizations} 考虑具有 Dirichlet 边界条件的 S-L 问题 \eqref{eq:sturm-liouville-inhomogeneous-differential-equation}. 设 $\lambda_n$ 为特征值, $\phi_n$ 为归一化特征函数. 设 $u\in L^2(a,b)$ 具有展开式 $u(x)=\sum_{n\in\mathbb{N}}\alpha_n\phi_n(x)$. 证明:\begin{enumerate}[label=(\alph*)]\item $u\in H^2(a,b)\cap H_0^1(a,b)$ 当且仅当 $\sum_{n\in\mathbb{N}}(1+\lambda_n^2)|\alpha_n^2|<\infty$.\item $u\in H_0^1(a,b)$ 当且仅当 $\sum_{n\in\mathbb{N}}(1+|\lambda_n|)|\alpha_n|^2<\infty$.\end{enumerate}(b) 的 Hint: 考虑内积 $(u,Lu)_w$.

\item\label{ex:sturm-liouville-smallest-eigenvalue-variational-characterization} 令
\begin{equation}\label{eq:sturm-liouville-first-eigenvalue-variational-problem}
\lambda=\min_{\substack{u\in H_0^1(a,b)\\\|u\|_w=1}}\int_a^bp(x)u'(x)^2+q(x)u(x)^2\,\mathrm{d}x.
\end{equation}
证明 $\lambda$ 是具有 Dirichlet 边界条件的 S-L 问题 \eqref{eq:sturm-liouville-inhomogeneous-differential-equation} 的最小特征值.

\item\label{ex:sturm-liouville-variational-eigenvalue-bounds} 显然, 前一题中对 $\lambda$ 的刻画可用于导出 $\lambda$ 的上下界. 导出一些这样的界.

\end{enumerate}
