\subsection{技术性结果}\label{sec:sobolev-technical-results}

\subsubsection{稠密性定理}\label{subsec:sobolev-density-theorems}

本小节证明具有有界支集的 $C^{\infty}$ 函数在 $H^k(\Omega)$ 中稠密. 不需要对边界正则性作任何假设. 若 $p<\infty$, 同样的证明适用于 $W^{k,p}(\Omega)$. 先证明有界支集函数稠密; 当然, 只有当 $\Omega$ 无界时这一点才值得关注.

\begin{Lemma}
\label{lem:sobolev-bounded-support-density}
具有有界支集的函数在 $H^k(\Omega)$ 中稠密.
\end{Lemma}

\begingroup
\postdisplaypenalty=10000
\renewcommand{\endProof}{\par\nobreak\hfill$\blacksquare$\par\vspace{0.1cm}}
\begin{Proof}
设 $\phi\in\mathcal{D}(\mathbb{R}^m)$ 在 $|\bm{x}|\leq1$ 时等于 $1$, 在 $|\bm{x}|\geq2$ 时等于 $0$, 在其余地方取介于两者之间的值. 对任意 $u\in H^k(\Omega)$, 考察序列 $u_n(\bm{x})=u(\bm{x})\phi_n(\bm{x})$, 其中 $\phi_n(\bm{x})=\phi(\bm{x}/n)$. 先证明 $u_n\to u$ 在 $L^2(\Omega)$ 中成立. 给定 $\varepsilon>0$, 设 $f$ 为满足 $\|u-f\|_2\leq\varepsilon$ 的连续函数. 得到
\begin{equation}\label{eq:sobolev-bounded-support-cutoff-convergence}
\|u\phi_n-u\|_2\leq\|(u-f)\phi_n\|_2+\|u-f\|_2+\|f-f\phi_n\|_2\leq2\varepsilon+\|f-f\phi_n\|_2.
\end{equation}
容易看出, 当 $n\to\infty$ 时 $f\phi_n$ 收敛到 $f$. 因而 $u\phi_n$ 在 $L^2$ 中收敛到 $u$. 导数的收敛由求导的乘积法则和直接的自举论证得到.
\end{Proof}
\endgroup

\begin{Theorem}
\label{thm:sobolev-interior-smooth-density}
$C^{\infty}(\Omega)\cap H^k(\Omega)$ 在 $H^k(\Omega)$ 中稠密.
\end{Theorem}

\begingroup
\postdisplaypenalty=10000
\renewcommand{\endProof}{\par\nobreak\hfill$\blacksquare$\par\vspace{0.1cm}}
\begin{Proof}
设 $\{\phi_i\}_{i\in\mathbb{N}}$ 为 $\Omega$ 上从属于局部有限覆盖 $\{U_i\}_{i\in\mathbb{N}}$ 的单位分解. 那么对任意 $u\in H^k(\Omega)$, 有 $u=\sum_{i\in\mathbb{N}}u\phi_i$ (在分布意义下, \emph{并非}在 $H^k(\Omega)$ 中收敛的意义下!). 此外, $u\phi_i$ 属于 $H^k(\Omega)$, 并且断言它实际上属于 $H_0^k(\Omega)$. 事实上, 可在 $\Omega$ 外将 $u\phi_i$ 零延拓, 得到 $H^k(\mathbb{R}^m)$ 中的函数. 由 Corollary~\ref{cor:full-space-test-functions-sobolev-density}, 存在序列 $f_n\in\mathcal{D}(\mathbb{R}^m)$, 使 $f_n\to u\phi_i$ 在 $H^k(\mathbb{R}^m)$ 中成立. 现在设试验函数 $\psi$ 在 $\phi_i$ 的支集上等于 $1$, 在 $U_i$ 外等于 $0$; 则有 $f_n\psi\to u\phi_i\psi=u\phi_i$, 且 $f_n\psi\in\mathcal{D}(\Omega)$.

因而对每个 $i\in\mathbb{N}$, 可以选择 $u_i^{(n)}\in\mathcal{D}(\Omega)$, 使
\begin{equation}\label{eq:sobolev-smooth-density-local-approximation-error}
\|u_i^{(n)}-u\phi_i\|_{k,2}\leq\dfrac{1}{n2^i}.
\end{equation}
于是
\begin{equation}\label{eq:sobolev-smooth-density-summed-approximation}
\sum_{i\in\mathbb{N}}u_i^{(n)}-u\phi_i=\sum_{i\in\mathbb{N}}u_i^{(n)}-u
\end{equation}
属于 $H^k(\Omega)$, 且范数 $\leq1/n$. 因而 $u^{(n)}:=\sum_{i\in\mathbb{N}}u_i^{(n)}$ 在 $H^k(\Omega)$ 中收敛到 $u$, 并且由于局部有限性, $u^{(n)}$ 为 $C^{\infty}$ 类.
\end{Proof}
\endgroup

因此, $H^k(\Omega)$ 中任意函数都可由导数在经典意义下存在的函数逼近. 这在证明中往往很有用. 可以先对光滑函数完成全部论证, 再通过取极限讨论一般情形. 然而, Theorem~\ref{thm:sobolev-interior-smooth-density} 仍不令人满意. 在许多情况下, 希望由直到边界都光滑的函数逼近, 而不仅仅在 $\Omega$ 内部光滑. 若不对 $\Omega$ 的边界作某些假设, 就无法做到这一点. 现在给出的结果并非最优, 但足以满足这里的需要.

\begin{Lemma}
\label{lem:sobolev-extension-boundary-smooth-density}
假设 $\Omega$ 具有 $k$ 阶延拓性质. 那么 $C^{\infty}(\overline\Omega)$ 中具有有界支集的函数在 $H^k(\Omega)$ 中稠密.
\end{Lemma}

证明由 Corollary~\ref{cor:full-space-test-functions-sobolev-density} 立即得到.
