\subsubsection{Gårding 不等式}\label{subsec:elliptic-garding-inequality}

现在证明椭圆 Dirichlet 问题的基本能量估计或强制性估计.

\begin{Theorem}{Gårding 不等式}
\label{thm:garding-elliptic-energy-inequality}
设 $\Omega$ 为具有 $k$ 阶延拓性质的有界区域, $L(\bm{x},D)$ 为 $2k$ 阶散度形式线性偏微分算子, 且以某个 $\theta>0$ 满足一致椭圆性条件 \eqref{eq:divergence-form-uniform-ellipticity-condition}. 再假设
\begin{equation}\label{eq:garding-principal-coefficient-continuity}
a_{\sigma\gamma}\in C_b(\overline\Omega)\quad\text{for all }|\sigma|=|\gamma|=k
\end{equation}
且
\begin{equation}\label{eq:garding-lower-coefficient-boundedness}
a_{\sigma\gamma}\in L^\infty(\Omega)\quad\text{for all }|\sigma|,|\gamma|\leq k.
\end{equation}
那么存在常数 $c_3$ 和 $\lambda_G\geq0$, 使得
\begin{equation}\label{eq:garding-elliptic-energy-inequality}
B[u,u]+\lambda\|u\|_{L^2(\Omega)}^2\geq c_3\|u\|_{H^k(\Omega)}^2\quad\text{for all }u\in H_0^k(\Omega).
\end{equation}

\end{Theorem}

\begingroup
\renewcommand{\endProof}{\par}
\begin{Proof}
设 $u\in H_0^k(\Omega)$. 首先将 $B[u,u]$ 分成主部与低阶项; 即令
\begin{equation}\label{eq:garding-principal-lower-splitting}
B[u,u]=I_1+I_2,
\end{equation}
其中
\begin{equation}\label{eq:garding-principal-bilinear-part}
I_1:=\sum_{|\sigma|=|\gamma|=k}\int_\Omega a_{\sigma\gamma}(\bm{x})D^\gamma uD^\sigma u\,\mathrm{d}\bm{x},
\end{equation}

\begin{equation}\label{eq:garding-lower-order-bilinear-part}
I_2:=\sum_{\substack{0\leq|\sigma|,|\gamma|\leq k\\|\sigma|+|\gamma|<2k}}\int_\Omega a_{\sigma\gamma}(\bm{x})D^\gamma uD^\sigma u\,\mathrm{d}\bm{x}.
\end{equation}
先估计低阶项. 给定 $\varepsilon>0$.
\begin{equation}\label{eq:garding-lower-order-holder-young-estimate}
\begin{aligned}
|I_2|&\leq C\sum_{\substack{0\leq|\sigma|,|\gamma|\leq k\\|\sigma|+|\gamma|<2k}}\int_\Omega|D^\gamma u|\,|D^\sigma u|\,\mathrm{d}\bm{x}\\
&\leq C\sum_{\substack{0\leq|\sigma|,|\gamma|\leq k\\|\sigma|+|\gamma|<2k}}\|D^\gamma u\|_2\|D^\sigma u\|_2\\
&\leq C\|u\|_{k,2}\|u\|_{k-1,2},\\
&\leq\dfrac{\varepsilon}{2}\|u\|_{k,2}^2+C(\varepsilon)\|u\|_{k-1,2}^2.
\end{aligned}
\end{equation}
这里使用了 Hölder 不等式和初等不等式
\begin{equation}\label{eq:garding-elementary-young-product-inequality}
ab\leq\varepsilon a^2+\dfrac{1}{4\varepsilon}b^2.
\end{equation}
现在令 $\delta>0$. 利用 Ehrling 引理的抽象形式 (Theorem~\ref{thm:ehrling-compact-intermediate-norm}) 及前面的估计, 得到
\begin{equation}\label{eq:garding-lower-order-ehrling-estimate}
|I_2|\leq\dfrac{\varepsilon}{2}\|u\|_{k,2}^2+C(\varepsilon)\left[\delta\|u\|_{k,2}^2+c(\delta)\|u\|_2^2\right].
\end{equation}
现在, 对任意 $\varepsilon>0$, 可选取充分小的 $\delta=\delta(\varepsilon)>0$, 使得
\begin{equation}\label{eq:garding-small-parameter-choice}
C(\varepsilon)\delta\leq\varepsilon/2.
\end{equation}
将此与前一个不等式结合, 得到估计:
\begin{equation}\label{eq:garding-final-lower-order-estimate}
|I_2|\leq\varepsilon\|u\|_{k,2}^2+C(\varepsilon)\|u\|_2^2.
\end{equation}

\end{Proof}
