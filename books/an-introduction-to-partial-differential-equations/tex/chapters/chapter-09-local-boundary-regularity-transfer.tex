现在可以结合前面的引理得到以下结果.

\begin{Lemma}
\label{lem:local-boundary-regularity-transfer}
设 Theorem~\ref{thm:global-boundary-regularity} 的假设成立. 则对每个 $\overline{\bm{x}}\in\partial\Omega$, 都存在包含 $\overline{\bm{x}}$ 的开集 $\widetilde Q\subset\mathbb{R}^n$, 使 $u\in H^2(\widetilde Q\cap\Omega)$, 并且
\begin{equation}\label{eq:local-boundary-neighborhood-h2-estimate}
\|u\|_{H^2(\widetilde Q\cap\Omega)}\leq C(\|f\|_{L^2(\Omega)}+\|u\|_{H^1(\Omega)}).
\end{equation}

\end{Lemma}

\begingroup
\renewcommand{\endProof}{\par\nobreak\hfill$\blacksquare$\par\vspace{0.1cm}}
\begin{Proof}
对每个 $\overline{\bm{x}}\in\partial\Omega$, 如上定义 $\bm{x}$ 空间中的集合 $\Omega^{\prime}$, $\bm{y}$ 空间中的集合 $\Omega^{\prime\prime}$, 以及映射 $\Psi:\Omega^{\prime}\to\Omega^{\prime\prime}$ 和 $\Phi:\Omega^{\prime\prime}\to\Omega^{\prime}$ (参见 Figure~\ref{fig:boundary-straightening-original}). 取 $\overline R$ 使 $B_{\overline R}(0)\cap\{\bm{y}\mid y_n>0\}\subset\Omega^{\prime\prime}$, 并定义
\begin{equation}\label{eq:boundary-transfer-local-half-ball}
Q^+:=B_{\overline R}(0)\cap\{\bm{y}\mid y_n>0\},
\end{equation}

\begin{equation}\label{eq:boundary-transfer-original-coordinate-neighborhood}
\widetilde Q:=\Phi(B_{\overline R}(0)), 
\end{equation}

\begin{equation}\label{eq:boundary-transfer-original-coordinate-positive-neighborhood}
\widetilde Q^+:=\Phi(Q^+).
\end{equation}
现在利用 Lemmas~\ref{lem:flat-boundary-local-regularity} 和 \ref{lem:boundary-transformed-uniform-ellipticity} 得到
\begin{equation}\label{eq:boundary-transfer-transformed-h2-estimate}
\|\widetilde u\|_{H^2(Q^+)}\leq C(\|\widetilde f\|_{L^2(\Omega^{\prime\prime})}+\|\widetilde u\|_{H^1(\Omega^{\prime\prime})}).
\end{equation}
由形如 \eqref{eq:boundary-coordinate-l2-norm-comparison} 的不等式得到
\begin{equation}\label{eq:boundary-transfer-original-h2-estimate}
\|u\|_{H^2(\widetilde Q^+)}\leq C(\|f\|_{L^2(\Omega^{\prime})}+\|u\|_{H^1(\Omega^{\prime})}), 
\end{equation}
从而立即得到 \eqref{eq:local-boundary-neighborhood-h2-estimate}.
\end{Proof}
\endgroup
