\Needspace{5\baselineskip}\textbf{Problems}

\begin{enumerate}[label=\thesection.\arabic*.,ref=\thesection.\arabic*,start=48,series=book-problems]

\item\label{ex:sturm-liouville-neumann-robin-spectrum} Find the eigenvalues and eigenfunctions for the following boundary-value problem:\[\begin{aligned}
-u''-\lambda u&=0,\\
u'(0)&=0,\\
\cos\beta\,u(1)+\sin\beta\,u'(1)&=0.
\end{aligned}\]

\item\label{ex:sturm-liouville-weight-x-dirichlet-spectrum} Find the eigenvalues and eigenfunctions for the following boundary-value problem:\[\begin{aligned}
-(xu')'-\lambda xu&=0,\quad0<a<x<b;\\
u(a)&=0,\\
u(b)&=0.
\end{aligned}\]

\item\label{ex:sturm-liouville-general-energy-bound} Prove \eqref{eq:sturm-liouville-lower-energy-bound} for the case $\tan\alpha\in(-\infty,0)$.

\item\label{ex:sturm-liouville-spectral-sobolev-characterizations} Consider the S-L problem \eqref{eq:sturm-liouville-inhomogeneous-differential-equation} with Dirichlet boundary conditions. Let $\lambda_n$ be the eigenvalues and let $\phi_n$ be the normalized eigenfunctions. Let $u\in L^2(a,b)$ have the expansion $u(x)=\sum_{n\in\mathbb{N}}\alpha_n\phi_n(x)$. Prove that\begin{enumerate}[label=(\alph*)]\item $u\in H^2(a,b)\cap H_0^1(a,b)$ iff $\sum_{n\in\mathbb{N}}(1+\lambda_n^2)|\alpha_n^2|<\infty$.\item $u\in H_0^1(a,b)$ iff $\sum_{n\in\mathbb{N}}(1+|\lambda_n|)|\alpha_n|^2<\infty$.\end{enumerate}Hint for (b): Consider the inner product $(u,Lu)_w$.

\item\label{ex:sturm-liouville-smallest-eigenvalue-variational-characterization} Let
\begin{equation}\label{eq:sturm-liouville-first-eigenvalue-variational-problem}
\lambda=\min_{\substack{u\in H_0^1(a,b)\\\|u\|_w=1}}\int_a^bp(x)u'(x)^2+q(x)u(x)^2\,\mathrm{d}x.
\end{equation}
Prove that $\lambda$ is the smallest eigenvalue of the S-L problem \eqref{eq:sturm-liouville-inhomogeneous-differential-equation} with Dirichlet boundary conditions.

\item\label{ex:sturm-liouville-variational-eigenvalue-bounds} Obviously, the characterization of $\lambda$ in the preceding problem can be used to derive upper and lower bounds for $\lambda$. Derive some such bounds.

\end{enumerate}
