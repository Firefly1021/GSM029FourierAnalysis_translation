\paragraph*{Problems}

\begin{enumerate}[label=\thesection.\arabic*.,ref=\thesection.\arabic*,start=15,series=book-problems]

\item\label{ex:periodic-fourier-uniform-convergence} Let $f\in L^2(-1,1)$. Show that the Fourier series of $f$ given by Corollary~\ref{cor:fourier-complex-hilbert-basis} converges uniformly if $f'\in L^2(-1,1)$ and in addition $f(-1)=f(1)$.

\item\label{ex:fourier-sine-series-linear-function} Find the Fourier sine series of the function $f(x)=x$ on the interval $[0,1]$. Show that the series converges uniformly on $[0,1-\delta]$ for any $\delta>0$. Hint: Consider also the Fourier sine series for the function $g(x)=x(1+\cos\pi x)$.

\item\label{ex:fourier-sine-series-interior-uniform-convergence} Let $f\in C^1[0,1]$. Show that the Fourier sine series for $f$ converges uniformly except near the endpoints of the interval. Hint: Write $f$ as the sum of a function which vanishes at the endpoints and a function whose Fourier series you can compute explicitly.

\item\label{ex:classical-polynomials-orthogonality-normalization} Verify the orthogonality of the Chebyshev, Hermite and Laguerre polynomials and find the factors necessary to normalize them.

\item\label{ex:laguerre-generating-function} Verify \eqref{eq:laguerre-polynomial-generating-function}.

\item\label{ex:laguerre-hermite-completeness-details} Fill in the details for showing the completeness of the Laguerre and Hermite polynomials.

\item\label{ex:hilbert-series-convergence-criteria} Prove Lemma~\ref{lem:unordered-hilbert-series-convergence-criteria}.

\item\label{ex:positive-power-span-lp-two-density} Is the span of $x$, $x^2$, $x^3$, etc., dense in $L^2(0,1)$?

\item\label{ex:separable-infinite-hilbert-spaces-isometry} Prove that all separable, infinite-dimensional Hilbert spaces are isometric.

\end{enumerate}
