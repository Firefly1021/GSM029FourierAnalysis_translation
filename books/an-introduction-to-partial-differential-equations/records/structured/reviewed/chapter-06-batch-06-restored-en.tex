\paragraph*{Problems}

\begin{enumerate}[label=\thesection.\arabic*.,ref=\thesection.\arabic*,start=1,series=book-problems]

\item\label{ex:norm-induced-metric} Let $(X,\|\cdot\|)$ be a normed vector space. Show that the function $d(x,y)=\|x-y\|$ defines a metric.

\item\label{ex:completion-theorem-proof-details} Provide the details for the proof of Theorem~\ref{thm:normed-space-completion}.

\item\label{ex:bounded-continuous-derivatives-banach} Let $\Omega$ be an open set in $\mathbb{R}^m$ and let $C_b^k(\overline\Omega)$ be the set of all functions on $\overline\Omega$ which have continuous bounded derivatives up to order $k$. Define
\begin{equation}\label{eq:bounded-continuous-derivatives-sum-norm}
\|u\|=\sum_{|\alpha|\leq k}\sup_{\bm{x}\in\overline\Omega}|D^\alpha u(\bm{x})|.
\end{equation}
Show that $C_b^k(\overline\Omega)$ is a Banach space.

\item\label{ex:young-conjugate-exponents-inequality} Prove Lemma~\ref{lem:young-conjugate-exponents-scalar-inequality}.

\item\label{ex:lp-product-conjugate-exponent-variant} Let $f\in\widetilde L^p(\Omega)$, $g\in\widetilde L^q(\Omega)$, where $p,q\in(1,\infty)$ and $r^{-1}=p^{-1}+q^{-1}<1$. Show that $fg\in\widetilde L^r(\Omega)$.

\item\label{ex:continuous-lp-not-complete} Show that the space $\widetilde L^p(\Omega)$ is not complete.

\item\label{ex:bounded-domain-lq-lp-inclusion} Let $\Omega$ be bounded and $1\leq p<q<\infty$. Show that $L^q(\Omega)\subset L^p(\Omega)$, and that there exists a constant $C$ depending only on $\Omega$ such that $\|u\|_p\leq C\|u\|_q$ for all $u\in L^q(\Omega)$.

\item\label{ex:lp-two-product-epsilon-bound} Show that for every $u,v\in L^2(\Omega)$ and every $\epsilon>0$ we have \[\left|\int_\Omega uv\,\mathrm{d}\bm{x}\right|\leq\epsilon\|u\|_2^2+C(\epsilon)\|v\|_2^2,\] where $C(\epsilon):=(4\epsilon)^{-1}$.

\item\label{ex:test-functions-dense-lp} Prove that $\mathcal{D}(\Omega)$ is dense in $L^p(\Omega)$, $1\leq p<\infty$.

\item\label{ex:inner-product-joint-continuity} Let $H$ be an inner product space. Prove that the inner product is continuous on $H\times H$.

\item\label{ex:sequence-lp-two-cauchy-schwarz-triangle} State the specific form of the Cauchy-Schwarz and triangle inequalities for $\ell^2$ and $L^2(\Omega)$.

\item\label{ex:lp-two-separable} Prove that $L^2(\Omega)$ is separable. Hint: Use Problem~\ref{ex:test-functions-dense-lp}.

\item\label{ex:closed-subspace-double-orthogonal-complement} Prove that $(M^\perp)^\perp=M$ if $M$ is closed.

\item\label{ex:finite-dimensional-norms-equivalence} Prove that all norms on $\mathbb{R}^n$ are equivalent.

\end{enumerate}
