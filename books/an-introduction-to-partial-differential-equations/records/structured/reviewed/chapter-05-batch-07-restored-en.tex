\paragraph*{Problems}

\begin{enumerate}[label=\thesection.\arabic*.,ref=\thesection.\arabic*,start=1,series=book-problems]

\item\label{ex:bump-is-test-function} Show that $\phi_{\bm{y},\epsilon}\in\mathcal{D}(\mathbb{R}^m)$.

\item\label{ex:test-function-support-escape} Show that the sequence $\phi_n(\bm{x})=n^{-1}\phi_{0,\epsilon}(\bm{x})$ converges to zero in $\mathcal{D}(\mathbb{R}^m)$. Show that the sequence $\psi_n(\bm{x})=n^{-1}\phi_{0,\epsilon}(\bm{x}/n)$ converges to zero uniformly and so do all derivatives. Why does $\psi_n$ nevertheless not converge to zero in $\mathcal{D}(\mathbb{R}^m)$? Does it converge to zero in $\mathcal{S}(\mathbb{R}^m)$?

\item\label{ex:prove-test-function-uniform-approximation} Prove Theorem~\ref{thm:test-functions-uniform-approximation}.

\item\label{ex:regular-distributions-injective} Let $f$ and $g$ be two different functions in $C(\Omega)$. Show that they also differ as generalized functions.

\item\label{ex:delta-not-continuous-function} Show that the Dirac delta function cannot be identified with any continuous function.

\item\label{ex:distribution-periodicity-radial-symmetry} Explain what it means for a generalized function to be periodic or radially symmetric.

\item\label{ex:distribution-local-vanishing} Let $f$ be a generalized function on $\Omega$ and let $G$ be an open subset of $\Omega$. Assume that every point in $G$ has a neighborhood on which $f$ vanishes. Prove that $f$ vanishes on $G$. Hint: Use a partition of unity argument.

\item\label{ex:pairing-away-from-distribution-support} Prove that if $\phi$ vanishes in a neighborhood of the support of $f$, then $(f,\phi)=0$. Would it suffice if $\phi$ vanishes on the support of $f$?

\item\label{ex:principal-value-regularization} Show that \eqref{eq:principal-value-one-over-x-regularization} does indeed define a generalized function and that the definition does not depend on $\epsilon$. How can one define a regularization of $1/x^2$?

\item\label{ex:compactly-supported-distribution-density} Prove that every distribution is the limit of a sequence of distributions with compact support. Hint: Let $f_n=f\psi_n$, where $\psi_n$ is a $C^\infty$ cutoff function.

\item\label{ex:distribution-multiplication-not-continuous} Show that \[\lim_{n\to\infty}\sin(nx)=0\] in $\mathcal{D}^{\prime}(\mathbb{R})$, but that \[\lim_{n\to\infty}\sin^2(nx)\ne0.\] Hence multiplication of distributions is not a continuous operation even where it is defined.

\item\label{ex:delta-approximation-square-divergence} Let $f_n$ be the sequence defined by \eqref{eq:delta-approximating-rectangular-sequence}. Show that \[\lim_{n\to\infty}f_n^2\] does not exist in the sense of distributions. Show, however, that \[\lim_{n\to\infty}f_n^2-n\delta\] exists.

\item\label{ex:gaussian-distribution-limit} Find \[\lim_{n\to\infty}\sqrt n\exp(-nx^2)\] in the sense of distributions.

\item\label{ex:distribution-not-tempered-convergence} Exhibit a sequence in $\mathcal{S}^{\prime}(\mathbb{R})$ which converges to zero in $\mathcal{D}^{\prime}(\mathbb{R})$, but not in $\mathcal{S}^{\prime}(\mathbb{R})$.

\item\label{ex:rapid-tests-weighted-uniform-convergence} Show that the sequence $\phi_n$ converges in $\mathcal{S}(\mathbb{R}^m)$ if and only if $|\bm{x}|^kD^\alpha\phi_n(\bm{x})$ converges uniformly for every $k\in\mathbb{N}\cup\{0\}$ and every $\alpha$.

\item\label{ex:tempered-distributions-sequential-completeness} Show that $\mathcal{S}^{\prime}(\mathbb{R}^m)$ is sequentially complete.

\item\label{ex:distribution-gluing-open-cover} Let $U_i$, $i\in\mathbb{N}$ be open sets such that $\Omega=\bigcup_{i\in\mathbb{N}}U_i$. Let $f_i\in\mathcal{D}^{\prime}(U_i)$ be given such that $f_i$ and $f_j$ agree on $U_i\cap U_j$. Show that there exists $f\in\mathcal{D}^{\prime}(\Omega)$ such that $f=f_i$ on $U_i$.

\item\label{ex:distribution-test-sequence-pairing} Prove Lemma~\ref{lem:distribution-test-sequence-pairing}.

\item\label{ex:locally-finite-cover-refinement} (a) Let $\Omega$ be any open subset of $\mathbb{R}^m$. Show that a family of subsets with the properties of Definition~\ref{def:partition-of-unity} exists.

(b) Let $\{U_i\}$ be any countable covering of $\Omega$ by open sets. A refinement of $\{U_i\}$ is a covering by open sets $V_k$, where each $V_k$ is a subset of one of the $U_i$. Given any covering of $\Omega$ by open sets, show that there is a refinement satisfying the properties of Definition~\ref{def:partition-of-unity}.

\end{enumerate}
