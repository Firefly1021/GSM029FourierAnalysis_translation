\paragraph*{Problems}

\begin{enumerate}[label=\thesection.\arabic*.,ref=\thesection.\arabic*,start=56,series=book-problems]

\item\label{ex:fourth-order-drift-adjoint-boundary} On the interval $[0,1]$, let $Lu=u''''+u'$ with boundary conditions $u''+u=u-u'''=0$ at the endpoints. Find the adjoint operator and the adjoint boundary conditions.

\item\label{ex:clamped-fourth-derivative-green-function} Find the Green\textquotesingle s function for the fourth derivative operator on $(0,1)$ with boundary conditions $u(0)=u^{\prime}(0)=u(1)=u^{\prime}(1)=0$.

\item\label{ex:biharmonic-normal-tangential-adjoint} Let $\Omega$ be a domain in $\mathbb{R}^2$ bounded by a smooth curve. Consider the equation $\Delta\Delta u=f$ with boundary conditions $\Delta u=\dfrac{\partial u}{\partial n}+\dfrac{\partial u}{\partial s}=0$. Determine the adjoint boundary-value problem.

\item\label{ex:evolution-initial-boundary-green-function} Define a Green\textquotesingle s function for an initial/boundary-value problem of the form $u_t=L(\bm{x},D)u+f$ with boundary conditions $B_j(\bm{x},D)u=0$. Give an analogue of the discussion in Section~\ref{subsec:green-functions-boundary-problems}.

\item\label{ex:verify-ball-green-normal-derivative} Verify \eqref{eq:ball-green-boundary-normal-derivative}.

\item\label{ex:neumann-boundary-integral-equation} Reformulate the Neumann problem as an integral equation on the boundary.

\end{enumerate}
