\paragraph*{Problems}

\begin{enumerate}[label=\thesection.\arabic*.,ref=\thesection.\arabic*,start=24,series=book-problems]

\item\label{ex:bounded-operator-space-completeness} Prove Theorem~\ref{thm:bounded-operator-space-completeness}.

\item\label{ex:hahn-banach-hilbert-extension-details} Verify the remarks at the end of Section~\ref{subsec:hahn-banach-theorem}.

\item\label{ex:nondense-linear-manifold-annihilating-functional} Let $M$ be a linear manifold in a Banach space $X$. Show that $M$ is not dense in $X$ if and only if there exists a nonzero $l\in X^*$ such that\[l(x)=0\quad\text{for every }x\in M.\]

\item\label{ex:infinite-hilbert-operator-space-nonseparable} Let $H$ be an infinite-dimensional Hilbert space. Prove that $\mathcal{L}(H)$ is not separable.

\item\label{ex:orthonormal-basis-weak-zero} Verify the claim of Example~\ref{ex:orthonormal-basis-weak-zero-example}.

\item\label{ex:bounded-linear-map-weak-continuity} Let $X,Y$ be Banach spaces and let $T\in\mathcal{L}(X,Y)$ and let $x_n$ be a weakly convergent sequence in $X$. Show that $Tx_n$ converges weakly in $Y$.

\item\label{ex:closed-linear-subspace-weak-sequential-closure} Let $M$ be a closed subspace of the Banach space $X$ and let $x_n\in M$ converge weakly to $x$. Show that $x\in M$. Hint: Show first that if $f(x)=0$ for every $f\in X^*$ with $f|_M=0$, then $x\in M$.

\item\label{ex:hilbert-weak-convergence-and-norms-imply-strong} Let $H$ be a Hilbert space. Let $x_n$ converge weakly to $x$ and assume in addition that $\|x_n\|\to\|x\|$. Show that $x_n\to x$ in norm (strongly).

\item\label{ex:bounded-l-two-distribution-limit-is-weak-limit} Let $f_n$ be a bounded sequence in $L^2(\Omega)$, and assume that $f_n$ converges in the sense of distributions. Show that $f_n$ converges weakly.

\item\label{ex:weak-strong-holder-dual-product-distribution-limit} Let $p\in(1,\infty)$, $q=p/(p-1)$. Suppose\[f_n\rightharpoonup\bar f\quad\text{(weakly) in }L^p(\Omega),\]\[g_n\to\bar g\quad\text{(strongly) in }L^q(\Omega).\]Show that\[f_ng_n\to\bar f\bar g\quad\text{in }\mathcal{D}^{\prime}(\Omega).\]

\item\label{ex:dirac-not-l-one-and-l-one-nonreflexivity} (a) Show that $\delta(x)\notin L^1(\mathbb{R})$. Hint: For given $g\in\widetilde L^1(\mathbb{R})$, find a continuous function $f$ with
\begin{equation}\label{eq:dirac-not-l-one-continuous-test-separation}
\left|\int_{-\infty}^{\infty}f(x)g(x)\,\mathrm{d}x-f(0)\right|\geq\dfrac{1}{2}\max_{x\in\mathbb{R}}|f(x)|.
\end{equation}
(b) Show that $L^1(\mathbb{R})$ is not reflexive.

\end{enumerate}
