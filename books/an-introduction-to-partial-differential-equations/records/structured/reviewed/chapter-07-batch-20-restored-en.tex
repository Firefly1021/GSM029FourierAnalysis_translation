\subsubsection{Problems}\label{subsec:sobolev-problems}

\begin{enumerate}[label=\thesection.\arabic*.,ref=\thesection.\arabic*,start=1,series=book-problems]

\item\label{ex:sobolev-difference-quotient-l-two-convergence} Let $u\in H^1(\mathbb{R})$. Show that $u'(x)=\lim_{h\to0}(u(x+h)-u(x))/h$ in the sense of $L^2$-convergence.

\item\label{ex:sobolev-line-integration-by-parts} Let $u,v\in H^1(\mathbb{R})$. Prove that\[\int_{-\infty}^{\infty}u(x)v'(x)\,\mathrm{d}x=-\int_{-\infty}^{\infty}u'(x)v(x)\,\mathrm{d}x.\]

\item\label{ex:compact-support-zero-boundary-sobolev} Assume $u\in H^k(\Omega)$ has compact support in $\Omega$. Prove that $u\in H_0^k(\Omega)$.

\item\label{ex:sobolev-embedding-critical-counterexample} Show that $H^1(\mathbb{R}^2)$ is not a subset of $C(\mathbb{R}^2)$. Hint: Consider powers of $|\ln|\bm{x}||$.

\item\label{ex:one-dimensional-h-one-product} Assume $u,v\in H^1(\mathbb{R})$. Show that the product $uv$ is also in $H^1(\mathbb{R})$.

\item\label{ex:sobolev-embedding-cusp-failure} Show that the Sobolev imbedding theorem fails in general if $\partial\Omega$ has a cusp. Hint: Consider functions with a singularity at the cusp.

\item\label{ex:sobolev-continuous-functions-vanish-at-infinity} Let $u\in H^s(\mathbb{R}^m)$ with $s>m/2$. Show that $\lim_{\bm{x}\to\infty}u(\bm{x})=0$.

\item\label{ex:weak-sobolev-convergence-lower-order-strong} Let $\Omega$ satisfy the assumptions of Theorem~\ref{thm:compact-successive-sobolev-orders}. Show that every weakly convergent sequence in $H^{k+1}(\Omega)$ converges strongly in $H^k(\Omega)$.

\item\label{ex:ehrling-sobolev-equivalent-norms} Prove Corollary~\ref{cor:ehrling-sobolev-equivalent-norms}.

\item\label{ex:zero-boundary-compact-sobolev-embedding-no-regularity} Let $\Omega$ be bounded (no assumptions on $\partial\Omega$). Prove that $H_0^{k+1}(\Omega)$ is compactly imbedded in $H^k(\Omega)$.

\item\label{ex:continuous-function-trace-support-lemma-proof} Prove Lemma~\ref{lem:continuous-function-trace-and-support}.

\item\label{ex:trace-without-half-order-loss-impossible} Show that there is no continuous trace operator mapping $H^k(\mathbb{R}^m)$ to $H^k(\mathbb{R}^{m-1})$.

\item\label{ex:one-direction-bounded-domain-weak-dirichlet-existence} Assume that $\Omega$ is bounded in one direction. Establish an existence result for a weak solution of the Dirichlet problem $\Delta u=f$, $u|_{\partial\Omega}=0$.

\item\label{ex:distribution-local-negative-sobolev-order} Let $D$ be open and bounded such that $\overline D\subset\Omega$ and let $f\in\mathcal{D}^{\prime}(\Omega)$. Show that there exists an integer $k$ such that $f|_D\in H^{-k}(D)$. Hint: Use Lemma~\ref{def:distribution} and the Sobolev imbedding theorem.

\item\label{ex:mean-zero-sobolev-poincare-inequality} Assume that $\Omega$ is bounded, connected, and has the $1$-extension property. Let\[V=\left\{u\in H^1(\Omega)\ \middle|\ \int_{\Omega}u(\bm{x})\,\mathrm{d}\bm{x}=0\right\}.\]Show that there is a constant $C$ such that\[\int_{\Omega}|u(\bm{x})|^2\,\mathrm{d}\bm{x}\leq C\int_{\Omega}|\nabla u(\bm{x})|^2\,\mathrm{d}\bm{x}\quad\text{for all }u\in V.\]For what other subsets $V$ of $H^1(\Omega)$ will such an inequality hold?

\item\label{ex:l-two-into-negative-one-compact-embedding} Show that if $\Omega$ is bounded, then $L^2(\Omega)$ is compactly embedded in $H^{-1}(\Omega)$.

\item\label{ex:whole-space-poincare-no-uniform-constant} Show that there is no constant $C$ such that\[\int_{\Omega}|u(\bm{x})|^2\,\mathrm{d}\bm{x}\leq C\int_{\Omega}|\nabla u(\bm{x})|^2\,\mathrm{d}\bm{x}\quad\text{for all }u\in H^1(\mathbb{R}^m).\]

\item\label{ex:narrow-unbounded-domain-smooth-not-l-p-example} Let\[\Omega=\{(x_1,x_2)\in\mathbb{R}^2\mid0<x_1<\infty,\quad0<x_2<e^{-x_1}\}.\]Show that for each $p\in(2,\infty)$, there is a function $u_p$ such that $u_p\in H^k(\Omega)$ for all $k\geq0$, but $u_p\notin L^p(\Omega)$.

\item\label{ex:titchmarsh-fourier-l-p-sobolev-embedding} A classical theorem of Titchmarsh asserts that if $p\in[1,2)$, then the Fourier transform maps $L^p(\mathbb{R}^m)$ into $L^q(\mathbb{R}^m)$ where $\dfrac{1}{p}+\dfrac{1}{q}=1$. Use this result to show that $H^1(\mathbb{R}^3)$ is continuously embedded in $L^p(\mathbb{R}^3)$ for all $p\in[2,6)$. (Note: $H^1(\mathbb{R}^3)$ is also embedded continuously in $L^6(\mathbb{R}^3)$.)

\item\label{ex:periodic-sobolev-spaces-fourier-characterization} Define Sobolev spaces of periodic functions on $\mathbb{R}$ and characterize them in terms of Fourier series. How are Sobolev spaces of periodic functions related to the Sobolev spaces on $[0,2\pi]$? Hint: Recall Problem~\ref{ex:periodic-fourier-uniform-convergence}.

\item\label{ex:boundary-smooth-functions-not-dense-open-set-example} Give an example of an open set such that $H^1(\Omega)\cap C^{\infty}(\overline\Omega)$ is not dense in $H^1(\Omega)$.

\item\label{ex:k-diffeomorphism-definition-redundancies} Discuss possible redundancies in the definition of a $k$-diffeomorphism.

\item\label{ex:manifold-sobolev-inner-products-equivalence} Verify that all the inner products defined by \eqref{eq:manifold-sobolev-local-inner-product} are equivalent.

\item\label{ex:vandermonde-matrix-nonsingularity} Let $A_{ij}=a_j^i$, $i,j=0,\ldots,K$, where the $a_j$ are distinct real numbers. (Use the convention $0^0=1$.) Show that $\det\bm{A}\ne0$.

\end{enumerate}
