\renewenvironment{Proof}{\par}{\par\nobreak\hfill$\blacksquare$\par\vspace{0.1cm}}
\begin{Proof}
(Here, $\|\cdot\|_2=\|\cdot\|_{L^2(D^+)}$, etc.) We get the following:\[\begin{aligned}
|I_1|&\leq\sum_i\|b_i\|_\infty\|\nabla u\|_2\|v\|_2+\|c\|_\infty\|u\|_2\|v\|_2+\|f\|_2\|v\|_2\\
&\leq C(\|u\|_{1,2}+\|f\|_2)\|w\|_{1,2}.\\[1ex]
|I_2|&\leq\sum_{i,j}\|a_{ij}\|_\infty\|D_k^hu\|_2\|w_{x_i}\|_2\|\nabla\zeta\|_\infty\\
&\leq C\|u\|_{1,2}\|w\|_{1,2}.\\[1ex]
|I_3|&\leq\sum_{i,j}\|a_{ij}\|_\infty\|\zeta_{x_i}\|_\infty\|D_k^hu\|_2\|w\|_2\\
&\leq C\|u\|_{1,2}\|w\|_{1,2}.\\[1ex]
|I_4|&\leq\sum_{i,j}\|a_{ij}\|_{1,\infty}\|\zeta\|_{1,\infty}\|u\|_{1,2}\|w\|_{1,2}\\
&\leq C\|u\|_{1,2}\|w\|_{1,2}.
\end{aligned}\]Combining these estimates and the uniform ellipticity condition gives us
\begin{equation}\label{eq:flat-boundary-coercive-localized-gradient-estimate}
\theta\int_{D^+}|\nabla w|^2\,\mathrm{d}\bm{x}\leq-\int_{D^+}a_{ij}(\bm{x}+h\bm{e}_k)w_{x_i}w_{x_j}\,\mathrm{d}\bm{x}\leq C(\|u\|_{1,2}+\|f\|_2)\|w\|_{1,2}.
\end{equation}
It follows that
\begin{equation}\label{eq:flat-boundary-localized-gradient-l2-bound}
\int_{D^+}|\nabla w|^2\leq C(\|f\|_2^2+\|u\|_{1,2}^2).
\end{equation}
We can use this and the fact that $\zeta\equiv1$ on $Q^+$ to show that\[\begin{aligned}
\int_{Q^+}|D_k^h\nabla u|^2\,\mathrm{d}\bm{x}&=\int_{Q^+}|\nabla w|^2\,\mathrm{d}\bm{x}\\
&\leq\int_{D^+}|\nabla w|^2\,\mathrm{d}\bm{x}\\
&\leq C(\|f\|_2^2+\|u\|_{1,2}^2).
\end{aligned}\]Thus, using Lemma~\ref{lem:difference-quotient-weak-derivative-bound}, we can get an estimate for all second-order mixed partial derivatives except for $u_{x_nx_n}$; i.e., we have
\begin{equation}\label{eq:flat-boundary-mixed-second-derivative-bound}
\sum_{\substack{i,j=1\\i+j<2n}}^n\int_{Q^+}|u_{x_ix_j}|^2\,\mathrm{d}\bm{x}\leq C(\|f\|_2^2+\|u\|_{1,2}^2).
\end{equation}
To estimate $u_{x_nx_n}$ we proceed as follows. From the interior regularity result we know that the differential equation\[L(D,\bm{x})u=f\]is satisfied in the strong or $L^2$ sense. Since $a_{nn}\in W^{1,\infty}(\Omega)$, the Sobolev embedding theorem ensures that it is continuous. Furthermore, the strong ellipticity condition ensures that $a_{nn}\geq\theta>0$. Thus, we can divide the PDE by $a_{nn}$ and rearrange to get
\begin{equation}\label{eq:flat-boundary-normal-second-derivative-recovery}
u_{x_nx_n}=-\frac{1}{a_{nn}}\left[\sum_{\substack{i,j=1\\i+j<2n}}^na_{ij}u_{x_ix_j}-g\right]
\end{equation}
Note that we have shown that the right-hand side of this equation is in $L^2(Q^+)$; and in addition, we have the estimate
\begin{equation}\label{eq:flat-boundary-normal-second-derivative-bound}
\int_{Q^+}|u_{x_nx_n}|^2\,\mathrm{d}\bm{x}\leq C(\|f\|_2^2+\|u\|_{1,2}^2).
\end{equation}
Combining this with our previous results we get our final estimate \eqref{eq:flat-boundary-local-h2-estimate}.
\end{Proof}
