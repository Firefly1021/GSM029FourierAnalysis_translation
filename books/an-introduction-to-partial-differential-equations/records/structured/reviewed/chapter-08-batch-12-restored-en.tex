\textbf{Problems}

\begin{enumerate}[label=\thesection.\arabic*.,ref=\thesection.\arabic*,start=12,series=book-problems]

\item\label{ex:closed-graph-implies-bounded-inverse} Show that the closed graph theorem implies the bounded inverse theorem.

\item\label{ex:orthogonal-projection-open-map} Let $M$ be a closed subspace of a Hilbert space $H$. Without using the open mapping theorem, show that the orthogonal projection operator $P:H\to M$ maps open sets in $H$ to open sets in $M$.

\item\label{ex:surjective-hilbert-operator-null-complement-bijection} Let $A:H\to Y$ be a bounded linear operator from a Hilbert space $H$ onto a Banach space $Y$. Let $\widetilde A:\mathcal{N}(A)^\perp\to Y$ be the restriction of $A$ to the domain $\mathcal{N}(A)^\perp$. Show that $\widetilde A$ is a continuous bijection.

\item\label{ex:open-map-need-not-preserve-closed-sets} We call a mapping open if it maps every open set to an open set. Show that an open mapping need not map closed sets to closed sets.

\item\label{ex:incomplete-finite-sequence-bounded-inverse-counterexample} Let $X$ be the space of sequences $x=\{x_1,x_2,x_3,\ldots\}$ with only finitely many nonzero terms and norm\[\|x\|:=\sup_{i\in\mathbb{N}}|x_i|.\]Let $T:X\to X$ be defined by\[Tx:=\left\{x_1,\dfrac{x_2}{2},\dfrac{x_3}{3},\ldots\right\}.\]Show that $T$ is linear and bounded but that $T^{-1}$ is unbounded. Why does this not contradict the bounded inverse theorem?

\end{enumerate}
