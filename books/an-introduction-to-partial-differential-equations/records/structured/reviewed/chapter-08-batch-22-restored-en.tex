\subsubsection{Proof of the Bounded Inverse Theorem for Hilbert Spaces}\label{subsec:hilbert-bounded-inverse-proof}

In this section we prove the result promised in Section~\ref{sec:open-mapping-theorem}.

\begin{Theorem}
\label{thm:hilbert-bounded-inverse}
If $X$ and $Y$ are Hilbert spaces and $A$ is a continuous bijection from $X$ to $Y$, then the inverse of $A$ is bounded.
\end{Theorem}

\begingroup
\postdisplaypenalty=10000
\renewcommand{\endProof}{\par\nobreak\hfill$\blacksquare$\par\vspace{0.1cm}}
\begin{Proof}
Since $A=A^{**}$, Problem~\ref{ex:hilbert-adjoint-bounded-inverse-equivalence} implies that it is enough to show that $A^*$ has a bounded inverse. Since the kernel of $A$ is trivial, the range of $A^*$ is dense in $X$. Thus, it is enough to show that there exists $\delta>0$ such that
\begin{equation}\label{eq:hilbert-bounded-inverse-adjoint-lower-bound}
\|A^*y\|\geq\delta\|y\|
\end{equation}
for all $y\in Y$. Suppose not, then there exists a sequence $y^n$ such that
\begin{equation}\label{eq:hilbert-bounded-inverse-adjoint-normalized-sequence}
\|A^*y^n\|=1
\end{equation}
and
\begin{equation}\label{eq:hilbert-bounded-inverse-unbounded-sequence}
\|y^n\|\to\infty.
\end{equation}
But now, for any $f\in Y$ we use the fact that $A$ is onto and let $x$ be the solution of $Ax=f$. Then
\begin{equation}\label{eq:hilbert-bounded-inverse-weak-bound}
|(y^n,f)|=|(y^n,Ax)|=|(A^*y^n,x)|\leq\|x\|;
\end{equation}
i.e., the sequence $y^n$ is weakly bounded. By the uniform boundedness principle (Theorem~\ref{thm:uniform-boundedness}) $y^n$ must be bounded in norm, a contradiction.
\end{Proof}
\endgroup
