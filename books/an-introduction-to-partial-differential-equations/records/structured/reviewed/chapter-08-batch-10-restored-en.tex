\textbf{Problems}

\begin{enumerate}[label=\thesection.\arabic*.,ref=\thesection.\arabic*,start=1,series=book-problems]

\item\label{ex:bounded-operator-closable-nonclosed-range} Show that every bounded operator is closable, but that the range of a bounded linear operator need not be closed.

\item\label{ex:closed-bounded-below-operator-closed-range} Let $A\in\mathcal{L}(X,Y)$. We say that $A$ is \emph{bounded below} on a subspace $M\subset X$ if there exists a constant $k>0$ such that\[\|Ax\|\geq k\|x\|,\quad\text{for all }x\in M.\]Let $(\mathcal{D}(A),A)$ be an operator from $H$ to $H$. Show that if $A$ is closed and bounded below on $\mathcal{N}(A)^\perp\cap\mathcal{D}(A)$, then $\mathcal{R}(A)$ is closed.

\item\label{ex:surjective-bounded-below-inverse} Let $A\in\mathcal{L}(X,Y)$ be surjective. Show that if $A$ is bounded below on $X$, then $A^{-1}$ exists and is bounded.

\item\label{ex:inverse-linear-operator-null-space-criterion} Prove Theorem~\ref{thm:inverse-linear-operator-null-space}.

\item\label{ex:composition-bijective-operators-inverse} Let $X$, $Y$ and $Z$ be Banach spaces, and let $A:X\to Y$ and $B:Y\to Z$ be bijective operators. Let $BA:X\to Z$ be the composition of $A$ and $B$. Show that\[(BA)^{-1}=A^{-1}B^{-1}.\]

\item\label{ex:integration-operator-supremum-norm} Find the norm of the integration operator $\mathcal{I}$ defined in Example~\ref{ex:integration-operator-fixed-lower-endpoint}.

\item\label{ex:operator-graph-norm-completeness} For any linear operator $(\mathcal{D}(A),A)$ from $X$ to $Y$ we define the \textbf{graph norm} on $\mathcal{D}(A)\subset X$ by\[\|x\|_G:=\|x\|_X+\|A(x)\|_Y\quad\text{for any }x\in\mathcal{D}(A).\]\begin{enumerate}[label=(\alph*)]\item Show that this does indeed define a norm on $\mathcal{D}(A)$.\item Show that $(\mathcal{D}(A),A)$ is closed if and only if $\mathcal{D}(A)$ equipped with the graph norm is a Banach space.\item Every normed linear space has a completion (hence $\mathcal{D}(A)$ equipped with the graph norm does as well) yet not every operator is closable. Why is this not a contradiction?\end{enumerate}

\item\label{ex:inverse-of-closed-operator-closed} Show that if the inverse of a closed operator exists, it is closed.

\item\label{ex:closed-operator-null-space-closed} Show that the nullspace of a closed operator is closed.

\item\label{ex:closed-operator-compact-images-and-preimages} Let $X$ and $Y$ be Banach spaces and let $(\mathcal{D}(A),A)$ be a closed operator from $X$ to $Y$. Show that the image under $A$ of a compact set in $X$ is a closed set in $Y$. Show that the inverse image of a compact set in $Y$ is a closed set in $X$.

\item\label{ex:relative-bound-closed-operator-sum} Let $A$ and $B$ be operators from $X$ to $Y$. We say that $B$ is bounded relative to $A$ if $\mathcal{D}(B)\supseteq\mathcal{D}(A)$ and there exists $a,b>0$ such that\[\|Bx\|_Y\leq a\|Ax\|_Y+b\|x\|_X\]for every $x\in\mathcal{D}(A)$. Prove the following: If $A$ is closed and $B$ is bounded relative to $A$ with $a<1$, then $A+B$ (with domain $\mathcal{D}(A)$) is closed.

\end{enumerate}
