We can now put the previous lemmas together to get the following result.

\begin{Lemma}
\label{lem:local-boundary-regularity-transfer}
Let the hypotheses of Theorem~\ref{thm:global-boundary-regularity} be satisfied. Then for each $\overline{\bm{x}}\in\partial\Omega$ there exists an open set $\widetilde Q\subset\mathbb{R}^n$ containing $\overline{\bm{x}}$ such that $u\in H^2(\widetilde Q\cap\Omega)$, and furthermore
\begin{equation}\label{eq:local-boundary-neighborhood-h2-estimate}
\|u\|_{H^2(\widetilde Q\cap\Omega)}\leq C(\|f\|_{L^2(\Omega)}+\|u\|_{H^1(\Omega)}).
\end{equation}

\end{Lemma}

\begingroup
\renewcommand{\endProof}{\par\nobreak\hfill$\blacksquare$\par\vspace{0.1cm}}
\begin{Proof}
For each $\overline{\bm{x}}\in\partial\Omega$ we let the sets $\Omega^{\prime}$ in $\bm{x}$ space, $\Omega^{\prime\prime}$ in $\bm{y}$ space and the maps $\Psi:\Omega^{\prime}\to\Omega^{\prime\prime}$ and $\Phi:\Omega^{\prime\prime}\to\Omega^{\prime}$ be defined as above (cf. Figure~\ref{fig:boundary-straightening-original}). Let $\overline R$ be such that $B_{\overline R}(0)\cap\{\bm{y}\mid y_n>0\}\subset\Omega^{\prime\prime}$ and define
\begin{equation}\label{eq:boundary-transfer-local-half-ball}
Q^+:=B_{\overline R}(0)\cap\{\bm{y}\mid y_n>0\},
\end{equation}

\begin{equation}\label{eq:boundary-transfer-original-coordinate-neighborhood}
\widetilde Q:=\Phi(B_{\overline R}(0)), 
\end{equation}

\begin{equation}\label{eq:boundary-transfer-original-coordinate-positive-neighborhood}
\widetilde Q^+:=\Phi(Q^+).
\end{equation}
Now, we can use Lemmas~\ref{lem:flat-boundary-local-regularity} and \ref{lem:boundary-transformed-uniform-ellipticity} to get
\begin{equation}\label{eq:boundary-transfer-transformed-h2-estimate}
\|\widetilde u\|_{H^2(Q^+)}\leq C(\|\widetilde f\|_{L^2(\Omega^{\prime\prime})}+\|\widetilde u\|_{H^1(\Omega^{\prime\prime})}).
\end{equation}
From inequalities such as \eqref{eq:boundary-coordinate-l2-norm-comparison} we get
\begin{equation}\label{eq:boundary-transfer-original-h2-estimate}
\|u\|_{H^2(\widetilde Q^+)}\leq C(\|f\|_{L^2(\Omega^{\prime})}+\|u\|_{H^1(\Omega^{\prime})}), 
\end{equation}
which leads immediately to \eqref{eq:local-boundary-neighborhood-h2-estimate}.
\end{Proof}
