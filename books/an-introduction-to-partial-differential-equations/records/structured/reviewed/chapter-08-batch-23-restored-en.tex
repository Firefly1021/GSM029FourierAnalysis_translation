\Needspace{5\baselineskip}\textbf{Problems}

\begin{enumerate}[label=\thesection.\arabic*.,ref=\thesection.\arabic*,start=24,series=book-problems]

\item\label{ex:complex-matrix-banach-hilbert-adjoints} Let $A$ be an $m\times n$ complex matrix, and define an operator (also called $A$) from $\mathbb{C}^n\to\mathbb{C}^m$ by matrix multiplication. What is the relationship among the adjoint, the Hilbert adjoint of the operator $A$ and the matrix $A$?

\item\label{ex:hilbert-adjoint-product-reversal} If $A$ and $B$ are in $\mathcal{L}(H)$ show that\[(AB)^*=B^*A^*.\]

\item\label{ex:operator-extension-adjoint-reverse-extension} Show that if $(\mathcal{D}(B),B)$ is an extension of $(\mathcal{D}(A),A)$, then $(\mathcal{D}(A^\times),A^\times)$ is an extension of $(\mathcal{D}(B^\times),B^\times)$.

\item\label{ex:adjoint-closure-theorem-proof-completion} Complete the proof of Theorem~\ref{thm:adjoint-closedness-and-double-adjoint-closure}.

\item\label{ex:right-shift-hilbert-adjoint} Compute the Hilbert adjoint of the right shift operator $S_r$ defined in Example~\ref{ex:right-left-shifts-domain-dependent-inverses}.

\item\label{ex:second-derivative-symmetric-not-self-adjoint} Let $H=L^2(0,1)$ and let\[\mathcal{D}(A)=\{u\in H^2(0,1)\mid u(0)=u'(0)=u(1)=0\}.\]Here the boundary conditions are taken in the sense of trace. Define $A:\mathcal{D}(A)\to H$ by\[A(u):=-\dfrac{\mathrm{d}^2u}{\mathrm{d}x^2}.\]Find the Hilbert adjoint of $(\mathcal{D}(A),A)$. Is the operator symmetric, self-adjoint?

\item\label{ex:classical-laplacian-boundary-domains-symmetric} Show that $(\widetilde{\mathcal{D}}_D(\Delta),\Delta)$ and $(\widetilde{\mathcal{D}}_N(\Delta),\Delta)$ defined in Example~\ref{ex:laplacian-dirichlet-neumann-domains} are symmetric.

\item\label{ex:adjoint-deficiency-eigenvalue-proof} Prove Theorem~\ref{thm:adjoint-deficiency-conjugate-eigenvalue}.

\item\label{ex:hilbert-adjoint-product-norm-square} Let $A\in\mathcal{L}(H)$. Show that $\|A^*A\|=\|A\|^2$.

\item\label{ex:self-adjoint-spectral-radius-equals-norm} It can be shown that for an operator $A\in\mathcal{L}(X)$\[r_\sigma(A)=\lim_{n\to\infty}\|A^n\|^{1/n}.\]Use this fact and Problem~\ref{ex:hilbert-adjoint-product-norm-square} to show that if $A\in\mathcal{L}(H)$ is self-adjoint, then $r_\sigma(A)=\|A\|$.

\item\label{ex:commuting-operator-product-spectral-radius} Suppose $A,B\in\mathcal{L}(X)$ and that $AB=BA$. Show that\[r_\sigma(AB)\leq r_\sigma(A)r_\sigma(B).\]Show that the commutativity assumption in this result is essential.

\item\label{ex:symmetric-operator-closable} Show that every symmetric operator is closable.

\item\label{ex:hilbert-adjoint-bounded-inverse-equivalence} Let $X$ and $Y$ be Hilbert spaces and suppose $A\in\mathcal{L}(X,Y)$ is a bijection. Show that $A$ has a bounded inverse if and only if $A^*$ does.

\item\label{ex:essential-self-adjointness-deficiency-proof} Prove Theorem~\ref{thm:essential-self-adjointness-zero-deficiency}.

\item\label{ex:right-left-shift-spectra} Describe the spectra of the right and left shift operators described in Example~\ref{ex:right-left-shifts-domain-dependent-inverses}.

\end{enumerate}
