\paragraph*{Problems}

\begin{enumerate}[label=\thesection.\arabic*.,ref=\thesection.\arabic*,start=20,series=book-problems]

\item\label{ex:piecewise-smooth-distribution-derivative} Let $f$ be a piecewise continuous function with a piecewise continuous derivative. Describe the distributional derivative of $f$.

\item\label{ex:logarithm-distribution-derivative} Find the distributional derivative of the function $\ln|x|$.

\item\label{ex:riemann-integrable-traveling-wave} Let $u(x,t)=f(x+t)$, where $f$ is any locally Riemann integrable function on $\mathbb{R}$. Show that $u_{tt}=u_{xx}$ in the sense of distributions.

\item\label{ex:inverse-square-distribution-laplacian} Evaluate $\Delta(1/r^2)$ in $\mathbb{R}^3$.

\item\label{ex:exponential-oscillation-tempered} Show that $e^x\cos e^x\in\mathcal{S}^{\prime}(\mathbb{R})$.

\item\label{ex:polynomial-cosine-coefficients-distribution} Show that $\sum_{n\in\mathbb{N}}a_n\cos nx$ converges in the sense of distributions, provided $|a_n|$ grows at most polynomially as $n\to\infty$.

\item\label{ex:prove-delta-primitive-conditions} Fill in the details for Example~\ref{ex:delta-sequence-primitive}.

\item\label{ex:distribution-ode-system-substitution} Discuss how the substitution \eqref{eq:distribution-ode-primitive-reduction} is generalized to systems of ODEs.

\item\label{ex:singular-ode-heaviside-general-solution} Show that the general solution of $xy'=0$ is $c_1+c_2H(x)$. Hint: Show first that if $\phi\in\mathcal{D}(\mathbb{R})$ vanishes at the origin, then $\phi(x)/x$ is a test function.

\item\label{ex:translation-invariant-distribution-constant} Let $f\in\mathcal{D}^{\prime}(\mathbb{R})$ be such that $f(x+h)=f(x)$ for every positive $h$. Show that $f$ is constant.

\item\label{ex:distribution-primitives-convergent-subsequence} Let $f_n$ be a convergent sequence in $\mathcal{D}^{\prime}(\mathbb{R})$ and assume that $F_n'=f_n$. Assume, in addition, that there is a test function $\phi_0$ with a nonzero integral such that the sequence $(F_n,\phi_0)$ is bounded. Show that $F_n$ has a convergent subsequence.

\item\label{ex:even-distribution-odd-primitive} Show that an even distribution on $\mathbb{R}$ has an odd primitive.

\item\label{ex:point-supported-distribution-delta-derivatives} Assume that the support of the distribution $f$ is the set $\{0\}$. Show that $f$ is a linear combination of derivatives of the delta function. Hint: Let $n$ be as given by Lemma~\ref{lem:distribution-local-finite-order} and assume that $D^\alpha\phi(0)$ vanishes for $|\alpha|\leq n$. Let $e$ be a test function which equals 1 for $|\bm{x}|\leq1$ and 0 for $|\bm{x}|\geq2$. Now consider the sequence $\phi_k(\bm{x})=\phi(\bm{x})e(k\bm{x})$. Show that $(f,\phi_k)\to0$ and hence $(f,\phi)=0$.

\end{enumerate}
