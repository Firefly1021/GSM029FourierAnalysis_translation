% Book-specific source-number and hyperlink-anchor control. The accepted
% environments and their rendered numbering remain unchanged.
\newcommand{\MBSourceChapterAnchor}{chapter-00}
\newcommand{\MBSetSourceChapterAnchor}[1]{%
  \renewcommand{\MBSourceChapterAnchor}{#1}%
  \renewcommand{\theHsection}{\MBSourceChapterAnchor.\arabic{section}}%
  \renewcommand{\theHsubsection}{\theHsection.\arabic{subsection}}%
  \renewcommand{\theHequation}{\MBSourceChapterAnchor.\arabic{section}.\arabic{equation}}%
  \renewcommand{\theHfigure}{\MBSourceChapterAnchor.figure.\arabic{figure}}%
  \renewcommand{\theHtable}{\MBSourceChapterAnchor.table.\arabic{table}}%
  \renewcommand{\theHthmcount}{\MBSourceChapterAnchor.statement.\arabic{section}.\arabic{thmcount}}%
}
\newcommand{\MBSourceStatementNumber}[2]{%
  \setcounter{thmcount}{\numexpr#2-1\relax}%
  \renewcommand{\theHthmcount}{\MBSourceChapterAnchor.#1.\arabic{section}.\arabic{thmcount}}%
}
