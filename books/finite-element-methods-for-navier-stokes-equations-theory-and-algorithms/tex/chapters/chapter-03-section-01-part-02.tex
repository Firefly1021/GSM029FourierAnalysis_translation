\subsection{抽象混合逼近}
\label{subsec:incompressible-abstract-mixed-discretization}

本小节始终假设 Theorem~\ref{thm:incompressible-weak-mixed-equivalence}
的条件成立.
对每个 $h$, 令 $X_h$ 和 $M_h$ 分别为
$\widetilde X$ 和 $\widetilde M$ 的两个有限维子空间.
用问题 $(Q_h)$ 逼近问题 $(\widetilde Q)$:

求 $(u_h,\lambda_h)\in X_h\times M_h$, 使
\begin{align}
\widetilde a(u_h,v_h)+\widetilde b(v_h,\lambda_h)
&=\langle l,v_h\rangle
&&\forall v_h\in X_h,
\label{eq:incompressible-discrete-q-first}\\
\widetilde b(u_h,\mu_h)
&=\langle\chi,\mu_h\rangle
&&\forall\mu_h\in M_h.
\label{eq:incompressible-discrete-q-second}
\end{align}
仍定义
\begin{equation}\label{eq:incompressible-discrete-affine-kernel}
\begin{aligned}
V_h(\chi)
&=\{v_h\in X_h;\
\widetilde b(v_h,\mu_h)=\langle\chi,\mu_h\rangle
\quad\forall\mu_h\in M_h\},\\
V_h&=V_h(0).
\end{aligned}
\end{equation}
接着, 与问题 $(Q_h)$ 相联系, 定义问题 $(P_h)$:

求 $u_h\in V_h(\chi)$, 使
\begin{equation}\label{eq:incompressible-discrete-p}
\widetilde a(u_h,v_h)=\langle l,v_h\rangle
\qquad\forall v_h\in V_h.
\end{equation}
这里 $V_h$ 通常也不包含在 $\widetilde V$ 中,
因此问题 $(P_h)$ 是问题 $(\widetilde P)$ 的外逼近.

为了导出 $u_h$ 和 $\lambda_h$ 的误差估计,
作与 \eqref{eq:incompressible-weak-v-ellipticity}
和 \eqref{eq:incompressible-restricted-weak-inf-sup} 类似的假设:

\textup{i)} 存在常数 $\alpha^*>0$, 使
\begin{equation}\label{eq:incompressible-discrete-vh-ellipticity}
\widetilde a(v_h,v_h)
\geq\alpha^*\|v_h\|_{\widetilde X}^2
\qquad\forall v_h\in V_h;
\end{equation}

\textup{ii)} 存在常数 $\beta^*>0$, 使
\begin{equation}\label{eq:incompressible-discrete-inf-sup}
\sup_{v_h\in X_h}
\frac{\widetilde b(v_h,\mu_h)}
{\|v_h\|_{\widetilde X}}
\geq\beta^*\|\mu_h\|_M
\qquad\forall\mu_h\in M_h.
\end{equation}

下一个 Theorem 是
Theorem~\ref{thm:primitive-abstract-discrete-error} 的自然推广.

\begin{Theorem}\label{thm:incompressible-abstract-mixed-error}
\textup{1)} 假设 $V_h(\chi)$ 非空, 且
$\widetilde a(\cdot,\cdot)$ 满足
\eqref{eq:incompressible-discrete-vh-ellipticity}.
则问题 $(P_h)$ 在 $V_h(\chi)$ 中有唯一解 $u_h$, 并且
\begin{equation}\label{eq:incompressible-velocity-abstract-error}
\begin{aligned}
\|u-u_h\|_{\widetilde X}
\leq{}&
\left(1+\frac{\|\widetilde a\|}{\alpha^*}\right)
\inf_{v_h\in V_h(\chi)}
\|u-v_h\|_{\widetilde X}\\
&+\frac1{\alpha^*}
\inf_{\mu_h\in M_h}
\left[
\sup_{v_h\in V_h}
\frac{\widetilde b(v_h,\lambda-\mu_h)}
{\|v_h\|_{\widetilde X}}
\right].
\end{aligned}
\end{equation}

\textup{2)} 再假设 $\widetilde b(\cdot,\cdot)$ 满足
\eqref{eq:incompressible-discrete-inf-sup}.
则 $V_h(\chi)$ 非空, 且问题 $(Q_h)$ 恰有一个解
$(u_h,\lambda_h)$, 其中 $u_h$ 是问题 $(P_h)$ 的解.
此外,
\begin{equation}\label{eq:incompressible-multiplier-abstract-error}
\|\lambda-\lambda_h\|_M
\leq
\frac{\|\widetilde a\|}{\beta^*}
\|u-u_h\|_{\widetilde X}
+
\inf_{\mu_h\in M_h}
\left\{
\frac{\|\widetilde b\|}{\beta^*}
\|\lambda-\mu_h\|_{\widetilde M}
+\|\lambda-\mu_h\|_M
\right\}.
\end{equation}
\end{Theorem}

\MathBookSourcePage{211}
\begin{Proof}
\textup{1)} 证明思路与
Theorem~\ref{thm:primitive-abstract-discrete-error} 非常相似.
当 $V_h(\chi)$ 非空时, 问题 $(P_h)$ 的解 $u_h$
的存在唯一性由
\eqref{eq:incompressible-discrete-vh-ellipticity}
和 Lax \& Milgram Theorem~\ref{thm:lax-milgram} 得到.

任取 $w_h\in V_h(\chi)$, 并令
$v_h=u_h-w_h\in V_h$.
则
\[
\widetilde a(v_h,v_h)
=\widetilde a(u-w_h,v_h)
+\widetilde b(v_h,\lambda-\mu_h)
\quad
\forall\mu_h\in M_h,\quad
\forall w_h\in V_h(\chi).
\]
因此 \eqref{eq:incompressible-discrete-vh-ellipticity} 蕴含
\[
\alpha^*\|v_h\|_{\widetilde X}
\leq
\|\widetilde a\|\,\|u-w_h\|_{\widetilde X}
+
\sup_{v_h\in V_h}
\frac{\widetilde b(v_h,\lambda-\mu_h)}
{\|v_h\|_{\widetilde X}}
\quad
\forall\mu_h\in M_h,\quad
\forall w_h\in V_h(\chi),
\]
这立即给出 \eqref{eq:incompressible-velocity-abstract-error}.

\textup{2)} 由于 $M_h$ 是有限维的,
\eqref{eq:incompressible-discrete-inf-sup}
蕴含 $M_h$ 上的经典 inf-sup 条件,
其常数最终可以依赖于 $h$.
因此 $V_h(\chi)$ 非空, 且问题 $(Q_h)$ 恰有一个解
$(u_h,\lambda_h)$, 其中 $u_h$ 满足问题 $(P_h)$.
此外, 对任意 $v_h\in X_h$ 和 $\mu_h\in M_h$, 有
\begin{equation}\label{eq:incompressible-discrete-multiplier-identity}
\widetilde b(v_h,\lambda_h-\mu_h)
=\widetilde a(u-u_h,v_h)
+\widetilde b(v_h,\lambda-\mu_h).
\end{equation}
于是由 \eqref{eq:incompressible-discrete-inf-sup} 得
\[
\beta^*\|\lambda_h-\mu_h\|_M
\leq
\|\widetilde a\|\,\|u-u_h\|_{\widetilde X}
+\|\widetilde b\|\,\|\lambda-\mu_h\|_{\widetilde M},
\]
从而得到
\eqref{eq:incompressible-multiplier-abstract-error}.
\end{Proof}

估计 \eqref{eq:incompressible-multiplier-abstract-error} 并非最优,
因为它用 $\|\lambda-\mu_h\|_{\widetilde M}$
给出 $\|\lambda-\lambda_h\|_M$ 的上界,
而这两个范数通常并非等价范数.
Section~\ref{sec:incompressible-further-sfvp}
中的例子将说明如何克服这一缺陷.

还要注意, 直接估计
\[
\inf_{v_h\in V_h(\chi)}
\|u-v_h\|_{\widetilde X}
\]
通常很困难.
与 Chapter~\ref{chap:stokes-primitive-variables} 中一样,
可以把这一项约化为 $X_h$ 中的逼近误差,
不过这里该过程并不总是最优的.
由于 $M_h$ 是有限维的, 存在常数 $K(h)>0$, 使
\begin{equation}\label{eq:incompressible-mh-norm-comparison}
\|\mu\|_{\widetilde M}
\leq K(h)\|\mu\|_M
\qquad\forall\mu\in M_h.
\end{equation}
由此容易证明下列结果.

\begin{Corollary}\label{cor:incompressible-abstract-mixed-quasioptimal}
在假设
\eqref{eq:incompressible-discrete-vh-ellipticity}
和 \eqref{eq:incompressible-discrete-inf-sup} 下,
每个 $v\in\widetilde V(\chi)$ 都满足
\begin{equation}\label{eq:incompressible-affine-space-approximation}
\inf_{v_h\in V_h(\chi)}
\|v-v_h\|_{\widetilde X}
\leq
\left[
1+K(h)\frac{\|\widetilde b\|}{\beta^*}
\right]
\inf_{v_h\in X_h}\|v-v_h\|_{\widetilde X}.
\end{equation}
此外, 存在仅依赖于
$\alpha^*,\beta^*,\|\widetilde a\|$ 和
$\|\widetilde b\|$ 的常数 $C>0$, 使
\begin{equation}\label{eq:incompressible-combined-abstract-error}
\begin{aligned}
\|u-u_h\|_{\widetilde X}
+\|\lambda-\lambda_h\|_M
\leq C\bigg\{&
[1+K(h)]
\inf_{v_h\in X_h}\|u-v_h\|_{\widetilde X}\\
&+\inf_{\mu_h\in M_h}
\|\lambda-\mu_h\|_{\widetilde M}
\bigg\}.
\end{aligned}
\end{equation}
\end{Corollary}

\MathBookSourcePage{212}
\begin{Proof}
显然,
\eqref{eq:incompressible-combined-abstract-error}
是 \eqref{eq:incompressible-affine-space-approximation}
和 Theorem~\ref{thm:incompressible-abstract-mixed-error}
的直接推论.

下面证明
\eqref{eq:incompressible-affine-space-approximation}.
由 \eqref{eq:incompressible-mh-norm-comparison},
inf-sup 条件
\eqref{eq:incompressible-discrete-inf-sup} 变为
\begin{equation}\label{eq:incompressible-scaled-discrete-inf-sup}
\sup_{v_h\in X_h}
\frac{\widetilde b(v_h,\mu_h)}
{\|v_h\|_{\widetilde X}}
\geq\frac{\beta^*}{K(h)}
\|\mu_h\|_{\widetilde M}
\qquad\forall\mu_h\in M_h.
\end{equation}
因此, 在
Theorem~\ref{thm:primitive-abstract-discrete-error}
的论证中用 $\beta^*/K(h)$ 代替 $\beta^*$,
即可直接得到
\eqref{eq:incompressible-affine-space-approximation}.
\end{Proof}

下一节的例子将表明, 当 $M_h$ 的维数趋于无穷时,
$K(h)$ 通常趋于无穷.
但是, Section~\ref{sec:incompressible-further-sfvp}
中发展的技术仍将说明, 在某些情形下如何不引入因子 $K(h)$
而精确估计 $V_h(\chi)$ 中的逼近误差.
