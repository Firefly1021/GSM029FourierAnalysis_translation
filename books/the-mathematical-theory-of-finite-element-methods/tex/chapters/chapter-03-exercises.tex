\label{exr:ch03-01}

\begin{MBExercise}

\begin{MBExercise}[非协调分片线性单元]

\begin{MBExercise}[旋转非协调双线性单元]
\label{exr:ch03-15}
令 $K=[-1,1]\times[-1,1]$, $\mathcal P$ 为由 $1,x,y,x^2-y^2$ 张成的形函数空间, $\mathcal N$ 由形函数在四条边中点处的值组成, 参见图~\ref{fig:ch03-non-element}. 证明 $(K,\mathcal P,\mathcal N)$ 是有限元. 令
\[
\MathBookFormulaID{formula-ch03-exercise-15-edge-mean}
a_j(v)=\frac1{|e_j|}\int_{e_j}v\,ds
\]
为 $v$ 在 $K$ 的边 $e_j$ 上的平均值, 并令 $\mathcal N^*=\{a_1,\ldots,a_4\}$. 证明 $(K,\mathcal P,\mathcal N^*)$ 也是有限元. 这两个单元是否插值等价?
\end{MBExercise}

\begin{MBExercise}[双三次 Hermite (Bogner--Fox--Schmit)单元]
\label{exr:ch03-16}
证明二元张量积三次多项式由下列量唯一确定:
\[
\MathBookFormulaID{formula-ch03-exercise-16-bogner-fox-schmit-nodes}
\Sigma=\left\{P(a_i),\frac{\partial P}{\partial x_1}(a_i),
\frac{\partial P}{\partial x_2}(a_i),
\frac{\partial^2P}{\partial x_1\partial x_2}(a_i):i=1,\ldots,4\right\},
\]
其中 $a_i$ 是矩形的顶点. 这是否在矩形剖分上生成 $C^1$ 分片三次函数?
\end{MBExercise}

\begin{MBExercise}
\label{exr:ch03-17}

\end{MBExercise}

\begin{MBExercise}
\label{exr:ch03-35}
设 $T$ 为顶点 $p_k$, $1\leq k\leq3$, 构成的三角形, $\lambda_j\in\mathcal P_1$ 满足 $\lambda_j(p_k)=\delta_{jk}$, $1\leq j\leq3$. 证明
\[
\MathBookFormulaID{formula-ch03-exercise-35-barycentric-integral}
\frac1{2|T|}\int_T\lambda_1^\ell\lambda_2^m\lambda_3^n\,dx
=\frac{\ell!m!n!}{(\ell+m+n+2)!},
\]
其中 $\ell,m,n$ 为非负整数. 四面体的对应公式是什么?
\end{MBExercise}
