\begin{MBExercise}

\[
\MathBookFormulaID{formula-ch11-ex07-closed-subspace-angle}
C(\|v_1\|+\|v_2\|)\leq\|v\|.
\]
提示: 使用开映射定理(Rudin 1987).
\end{MBExercise}

\begin{MBExercise}
\label{exr:ch11-23}
证明引理~\ref{lem:ch11-divergence-right-inverse} 的如下离散版本. 设 $\Omega$ 是凸多边形区域. 给定任意 $p\in L^2(\Omega)$, 使得对所有 $T\in\mathcal T^h$, $p|_T$ 都是常数且 $\int_\Omega p\,dx=0$, 存在正常数 $C$ 和 $\chvec v\in\chvec V_h^s$, 使得
\[
\MathBookFormulaID{formula-ch11-ex23-discrete-divergence-lifting}
\operatorname{div}_h\chvec v=p,
\qquad
\|\chten{\operatorname{grad}}_h\chvec v\|_{\chten L^2(\Omega)}
\leq C\|p\|_{L^2(\Omega)}.
\]
提示: 使用引理~\ref{lem:ch11-divergence-right-inverse}, 插值算子 $\Pi_h$ 和~\eqref{eq:ch11-interpolant-divergence-average}.
\end{MBExercise}

\begin{MBExercise}
\label{exr:ch11-24}
令 $\chvec V_h^s$ 由~\eqref{eq:ch11-nonconforming-vector-space} 定义. 子空间
\[
\MathBookFormulaID{formula-ch11-ex24-discrete-kernel}
\chvec W_h:=\{\chvec v\in\chvec V_h^s:\operatorname{div}_h\chvec v=0\}
\]
可用于不可压缩流体流动问题. 以 $\psi_e$ 表示在边 $e$ 的中点取值 $1$ 而在其他所有中点取值 $0$ 的分片线性函数. 第一类 $W_h$ 基函数与内边相关. 令 $e$ 是 $\mathcal T^h$ 的一条内边, $\chvec t_e$ 是 $e$ 的单位切向量, 则 $\chvec\phi_e:=\psi_e\chvec t_e$. 第二类基函数与内顶点相关. 令 $p$ 是一个内顶点, $e_1,e_2,\ldots,e_l$ 是以 $p$ 为端点的边, 则
\[
\MathBookFormulaID{formula-ch11-ex24-vertex-basis}
\chvec\phi_p:=\sum_{i=1}^l|e_i|\psi_{e_i}\chvec n_{e_i},
\]
其中 $\chvec n_{e_i}$ 是指向绕 $p$ 逆时针方向内部的 $e_i$ 的单位法向量(见图~\ref{fig:ch11-kernel-basis-function}). 证明 $W_h$ 的一组基由下列两个集合的并给出:
\[
\MathBookFormulaID{formula-ch11-ex24-edge-basis-set}
\{\chvec\phi_e:e\text{ 是 }\mathcal T^h\text{ 的内边}\},
\qquad
\MathBookFormulaID{formula-ch11-ex24-vertex-basis-set}
\{\chvec\phi_p:p\text{ 是 }\mathcal T^h\text{ 的内顶点}\}.
\]
提示: 必要时参见 Thomasset (1981).
\end{MBExercise}
